   312→   312→-- 7. AJOUT COLONNES SUBSCRIPTION AU PROFIL
   313→   313→-- ============================================================
   314→   314→-- On ajoute les colonnes au profil existant pour un acces rapide
   315→   315→-- (evite un JOIN avec subscriptions a chaque requete).
   316→   316→-- Ces colonnes sont des COPIES miroirs de l'abonnement actif.
   317→   317→
   318→   318→-- Colonne plan (supprime l'ancienne si elle existe pour la recreer proprement)
   319→   319→DO $$ BEGIN
   320→   320→  IF EXISTS (
   321→   321→    SELECT 1 FROM information_schema.columns
   322→   322→    WHERE table_schema = 'public' AND table_name = 'profiles' AND column_name = 'plan'
   323→   323→  ) THEN
   324→   324→    ALTER TABLE public.profiles DROP COLUMN plan;
   325→   325→  END IF;
   326→   326→END $$;
   327→   327→
   328→   328→ALTER TABLE public.profiles
   329→   329→  ADD COLUMN IF NOT EXISTS plan TEXT DEFAULT 'starter'
   330→   330→    CHECK (plan IN ('starter', 'pro', 'business')),
   331→   331→  ADD COLUMN IF NOT EXISTS subscription_status TEXT DEFAULT 'none'
   332→   332→    CHECK (subscription_status IN ('none', 'active', 'past_due', 'cancelled', 'expired')),
   333→   333→  ADD COLUMN IF NOT EXISTS subscription_start TIMESTAMPTZ,
   334→   334→  ADD COLUMN IF NOT EXISTS subscription_end TIMESTAMPTZ,
   335→   335→  ADD COLUMN IF NOT EXISTS subscription_id UUID REFERENCES public.subscriptions(id) ON DELETE SET NULL,
   336→   336→  ADD COLUMN IF NOT EXISTS chariow_customer_id TEXT;
   337→   337→
   338→   338→-- Renommer stripe_customer_id en chariow_customer_id si l'ancien existe
   339→   339→DO $$ BEGIN
   340→   340→  IF EXISTS (
   341→   341→    SELECT 1 FROM information_schema.columns
   342→   342→    WHERE table_schema = 'public' AND table_name = 'profiles'
   343→   343→      AND column_name = 'stripe_customer_id'
   344→   344→  ) THEN
   345→   345→    ALTER TABLE public.profiles RENAME COLUMN stripe_customer_id TO chariow_customer_id;
   346→   346→  END IF;
   347→   347→END $$;
   348→   348→
   349→   349→COMMENT ON COLUMN public.profiles.plan IS 'Plan actuel (miroir de la subscription active)';
   350→   350→COMMENT ON COLUMN public.profiles.subscription_status IS 'Statut de l\'abonnement actuel';
   351→   351→COMMENT ON COLUMN public.profiles.chariow_customer_id IS 'ID client chez Chariow (si applicable)';

   354→   354→-- 8. TRIGGER : Mise a jour miroir profil <-> subscription
   355→   355→-- ============================================================
   356→   356→
   357→   357→-- Quand une subscription est creee/activee, mettre a jour le profil
   358→   358→CREATE OR REPLACE FUNCTION public.sync_profile_subscription()
   359→   359→RETURNS TRIGGER AS $$
   360→   360→DECLARE
   361→   361→  v_plan_name TEXT;
   362→   362→  v_is_active BOOLEAN;
   363→   363→BEGIN
   364→   364→  -- Ne pas propager si on est deja en train de mettre a jour le profil
   365→   365→  -- (evite les boucles infinies avec le trigger ci-dessous)
   366→   366→  IF TG_OP = 'UPDATE' AND OLD.profile_id IS NOT DISTINCT FROM NEW.profile_id
   367→   367→     AND OLD.plan_id IS NOT DISTINCT FROM NEW.plan_id
   368→   368→     AND OLD.status IS NOT DISTINCT FROM NEW.status
   369→   369→     AND OLD.starts_at IS NOT DISTINCT FROM NEW.starts_at
   370→   370→     AND OLD.ends_at IS NOT DISTINCT FROM NEW.ends_at THEN
   371→   371→    RETURN NEW;
   372→   372→  END IF;
   373→   373→
   374→   374→  v_is_active := (NEW.status = 'active' AND NEW.ends_at > now());
   375→   375→
   376→   376→  IF v_is_active THEN
   377→   377→    v_plan_name := COALESCE(
   378→   378→      (SELECT name FROM public.plans WHERE id = NEW.plan_id),
   379→   379→      NEW.plan_name,
   380→   380→      NEW.plan_id
   381→   381→    );
   382→   382→
   383→   383→    UPDATE public.profiles SET
   384→   384→      plan            = NEW.plan_id,
   385→   385→      subscription_status = 'active',
   386→   386→      subscription_start   = NEW.starts_at,
   387→   387→      subscription_end     = NEW.ends_at,
   388→   388→      subscription_id      = NEW.id,
   389→   389→      updated_at           = now()
   390→   390→    WHERE id = NEW.profile_id;
   391→   391→  ELSE
   392→   392→    -- Si la subscription n'est plus active, retrograder le profil
   393→   393→    UPDATE public.profiles SET
   394→   394→      plan            = 'starter',
   395→   395→      subscription_status = COALESCE(
   396→   396→        CASE NEW.status
   397→   397→          WHEN 'cancelled' THEN 'cancelled'
   398→   398→          WHEN 'expired'  THEN 'expired'
   399→   399→          WHEN 'upgraded' THEN 'active'  -- Le nouveau plan est deja actif
   400→   400→          WHEN 'downgraded' THEN 'active' -- Le nouveau plan est deja actif
   401→   401→          ELSE 'none'
   402→   402→        END,
   403→   403→        'none'
   404→   404→      ),
   405→   405→      subscription_start   = NULL,
   406→   406→      subscription_end     = NULL,
   407→   407→      subscription_id      = NULL,
   408→   408→      updated_at           = now()
   409→   409→    WHERE id = NEW.profile_id
   410→   410→      AND subscription_id = NEW.id;  -- Ne mettre a jour que si c'est l'abonnement reference
   411→   411→  END IF;
   412→   412→
   413→   413→  RETURN NEW;
   414→   414→END;
   415→   415→$$ LANGUAGE plpgsql SECURITY DEFINER;
   416→   416→
   417→   417→DROP TRIGGER IF EXISTS trg_sync_profile_subscription ON public.subscriptions;
   418→   418→CREATE TRIGGER trg_sync_profile_subscription
   419→   419→  AFTER INSERT OR UPDATE OF status, ends_at, plan_id ON public.subscriptions
   420→   420→  FOR EACH ROW EXECUTE FUNCTION public.sync_profile_subscription();

   423→   423→-- 9. TRIGGER : Expiration automatique des abonnements
   424→   424→-- ============================================================
   425→   425→-- Fonction qui expire les abonnements dont la date de fin est passee.
   426→   426→-- A appeler via un cron job (GitHub Actions) ou manuellement.
   427→   427→
   428→   428→CREATE OR REPLACE FUNCTION public.expire_subscriptions()
   429→   429→RETURNS INTEGER AS $$
   430→   430→DECLARE
   431→   431→  v_count INTEGER;
   432→   432→BEGIN
   433→   433→  -- Verrouillage advisory pour eviter les executions concurrentes
   434→   434→  PERFORM pg_advisory_lock(54321);
   435→   435→
   436→   436→  UPDATE public.subscriptions

   485→   485→-- Incrementer un compteur d'usage mensuel
   486→   486→CREATE OR REPLACE FUNCTION public.increment_usage(
   487→   487→  p_profile_id UUID,
   488→   488→  p_counter TEXT   -- 'reminders_sent' ou 'appointments_created'
   489→   489→)
   490→   490→RETURNS INTEGER AS $$
   491→   491→DECLARE
   492→   492→  v_month TEXT := to_char(now(), 'YYYY-MM');
   493→   493→  v_new_count INTEGER;
   494→   494→BEGIN
   495→   495→  INSERT INTO public.subscription_usage (profile_id, usage_month, p_counter)
   496→   496→  VALUES (p_profile_id, v_month, 1)
   497→   497→  ON CONFLICT (profile_id, usage_month) DO UPDATE SET
   498→   498→    -- On ne peut pas utiliser un paramètre comme nom de colonne dynamiquement
   499→   499→    -- donc on fait un CASE
   500→   500→    reminders_sent = CASE
   501→   501→      WHEN p_counter = 'reminders_sent' THEN subscription_usage.reminders_sent + 1
   502→   502→      ELSE subscription_usage.reminders_sent
   503→   503→    END,
   504→   504→    appointments_created = CASE
   505→   505→      WHEN p_counter = 'appointments_created' THEN subscription_usage.appointments_created + 1
   506→   506→      ELSE subscription_usage.appointments_created
   507→   507→    END,
   508→   508→    updated_at = now()
   509→   509→  RETURNING
   510→   510→    CASE p_counter
   511→   511→      WHEN 'reminders_sent' THEN subscription_usage.reminders_sent
   512→   512→      ELSE subscription_usage.appointments_created
   513→   513→    END INTO v_new_count;
   514→   514→
   515→   515→  RETURN COALESCE(v_new_count, 0);
   516→   516→EXCEPTION WHEN OTHERS THEN
   517→   517→  RAISE WARNING 'increment_usage failed: %', SQLERRM;
   518→   518→  RETURN -1;
   519→   519→END;
   520→   520→$$ LANGUAGE plpgsql SECURITY DEFINER;

   522→   522→-- Verifier si un utilisateur peut utiliser une fonctionnalite
   523→   523→CREATE OR REPLACE FUNCTION public.can_use_feature(
   524→   524→  p_profile_id UUID,
   525→   525→  p_feature TEXT  -- 'professionals', 'services', 'reminders', 'appointments_per_day', 'clients'
   526→   526→)
   527→   527→RETURNS TABLE(allowed BOOLEAN, current_limit INTEGER, current_count INTEGER, plan_name TEXT) AS $$
   528→   528→DECLARE
   529→   529→  v_plan RECORD;
   530→   530→  v_current_count INTEGER := 0;
   531→   531→  v_limit INTEGER;
   532→   532→  v_month TEXT := to_char(now(), 'YYYY-MM');
   533→   533→BEGIN
   534→   534→  -- Recuperer le plan actuel du profil
   535→   535→  SELECT p.* INTO v_plan
   536→   536→  FROM public.plans p
   537→   537→  JOIN public.profiles pr ON pr.plan = p.id
   538→   538→  WHERE pr.id = p_profile_id;
   539→   539→
   540→   540→  IF NOT FOUND THEN
   541→   541→    -- Profil sans plan = starter par defaut (plan le plus bas)
   542→   542→    SELECT * INTO v_plan FROM public.plans WHERE id = 'starter' LIMIT 1;
   543→   543→  END IF;
   544→   544→
   545→   545→  -- Calculer la limite selon la fonctionnalite
   546→   546→  v_limit := CASE p_feature
   547→   547→    WHEN 'professionals'        THEN v_plan.max_professionals
   548→   548→    WHEN 'services'             THEN v_plan.max_services
   549→   549→    WHEN 'reminders'            THEN v_plan.max_reminders_per_month
   550→   550→    WHEN 'appointments_per_day' THEN v_plan.max_appointments_per_day
   551→   551→    WHEN 'clients'              THEN v_plan.max_clients
   552→   552→    ELSE 0
   553→   553→  END;
   554→   554→
   555→   555→  -- Calculer le compteur actuel
   556→   556→  IF p_feature = 'reminders' THEN
   557→   557→    SELECT COALESCE(SUM(reminders_sent), 0) INTO v_current_count
   558→   558→    FROM public.subscription_usage
   559→   559→    WHERE profile_id = p_profile_id AND usage_month = v_month;
   560→   560→  ELSIF p_feature = 'appointments_per_day' THEN
   561→   561→    SELECT COUNT(*) INTO v_current_count
   562→   562→    FROM public.appointments
   563→   563→    WHERE profile_id = p_profile_id
   564→   564→      AND date(starts_at) = CURRENT_DATE
   565→   565→      AND status != 'cancelled';
   566→   566→  ELSIF p_feature = 'services' THEN
   567→   567→    SELECT COUNT(*) INTO v_current_count
   568→   568→    FROM public.services
   569→   569→    WHERE profile_id = p_profile_id AND is_active = true;
   570→   570→  ELSIF p_feature = 'clients' THEN
   571→   571→    SELECT COUNT(*) INTO v_current_count
   572→   572→    FROM public.clients
   573→   573→    WHERE profile_id = p_profile_id;
   574→   574→  ELSIF p_feature = 'professionals' THEN
   575→   575→    -- Pour l'instant, 1 profil = 1 professionnel
   576→   576→    v_current_count := 1;
   577→   577→  END IF;
   578→   578→
   579→   579→  -- -1 = illimite
   580→   580→  allowed := (v_limit = -1) OR (v_current_count < v_limit);
   581→   581→
   582→   582→  RETURN QUERY SELECT allowed, v_limit, v_current_count, v_plan.name;
   583→   583→  RETURN;
   584→   584→END;
   585→   585→$$ LANGUAGE plpgsql SECURITY DEFINER STABLE;

   587→   587→-- Creer un abonnement (appele par le webhook apres paiement valide)
   588→   588→CREATE OR REPLACE FUNCTION public.activate_subscription(
   589→   589→  p_profile_id UUID,
   590→   590→  p_plan_id    TEXT,
   591→   591→  p_payment_id UUID DEFAULT NULL,
   592→   592→  p_duration_days INTEGER DEFAULT 30
   593→   593→)
   594→   594→RETURNS UUID AS $$
   595→   595→DECLARE
   596→   596→  v_existing_active UUID;
   597→   597→  v_new_sub_id UUID;
   598→   598→  v_old_status TEXT;
   599→   599→BEGIN
   600→   600→  -- Verrouillage advisory par profil (evite les doubles activations)
   601→   601→  PERFORM pg_advisory_lock(hashtext(p_profile_id::text));
   602→   602→
   603→   603→  -- Trouver l'abonnement actuel s'il existe
   604→   604→  SELECT id INTO v_existing_active
   605→   605→  FROM public.subscriptions
   606→   606→  WHERE profile_id = p_profile_id AND status = 'active';
   607→   607→
   608→   608→  IF v_existing_active IS NOT NULL THEN
   609→   609→    -- Determiner si c'est un upgrade ou downgrade
   610→   610→    SELECT
   611→   611→      CASE
   612→   612→        WHEN (SELECT tier_priority FROM plans WHERE id = p_plan_id) >
   613→   613→             (SELECT tier_priority FROM plans WHERE id = s.plan_id)
   614→   614→        THEN 'upgraded'
   615→   615→        ELSE 'downgraded'
   616→   616→      END INTO v_old_status
   617→   617→    FROM public.subscriptions s WHERE id = v_existing_active;
   618→   618→
   619→   619→    -- Desactiver l'ancien abonnement
   620→   620→    UPDATE public.subscriptions
   621→   621→    SET status = v_old_status, updated_at = now()
   622→   622→    WHERE id = v_existing_active;
   623→   623→  END IF;
   624→   624→
   625→   625→  -- Creer le nouvel abonnement
   626→   626→  INSERT INTO public.subscriptions (
   627→   627→    profile_id, plan_id, plan_name,
   628→   628→    starts_at, ends_at,
   629→   629→    status, payment_id
   630→   630→  ) VALUES (
   631→   631→    p_profile_id, p_plan_id,
   632→   632→    COALESCE((SELECT name FROM plans WHERE id = p_plan_id), p_plan_id),
   633→   633→    now(),
   634→   634→    now() + (p_duration_days || ' days')::INTERVAL,
   635→   635→    'active',
   636→   636→    p_payment_id
   637→   637→  )
   638→   638→  RETURNING id INTO v_new_sub_id;
   639→   639→
   640→   640→  -- Le trigger trg_sync_profile_subscription va automatiquement
   641→   641→  -- mettre a jour les colonnes du profil
   642→   642→
   643→   643→  -- Release le verrou
   644→   644→  PERFORM pg_advisory_unlock(hashtext(p_profile_id::text));
   645→   645→
   646→   646→  RETURN v_new_sub_id;
   647→   647→END;
   648→   648→$$ LANGUAGE plpgsql SECURITY DEFINER;
   649→   649→
   650→   650→COMMENT ON FUNCTION public.activate_subscription() IS 'Active un abonnement apres paiement. Desactive l\'ancien si existant. Verrouille par profil pour eviter les doubles activations.';

   873→   873→-- Vue : resume abonnement d'un profil
   874→   874→CREATE OR REPLACE VIEW public.v_subscription_summary AS
   875→   875→SELECT
   876→   876→  p.id AS profile_id,
   877→   877→  p.plan,
   878→   878→  p.subscription_status,
   879→   879→  p.subscription_start,
   880→   880→  p.subscription_end,
   881→   881→  pl.name        AS plan_name,
   882→   882→  pl.price_monthly,
   883→   883→  pl.features,
   884→   884→  CASE
   885→   885→    WHEN p.subscription_status = 'active' AND p.subscription_end > now()
   886→   886→    THEN GREATEST(0, CEIL(EXTRACT(EPOCH FROM (p.subscription_end - now())) / 86400))
   887→   887→    ELSE 0
   888→   888→  END AS days_remaining,
   889→   889→  CASE
   890→   890→    WHEN p.subscription_status = 'active' AND p.subscription_end > now() THEN true
   891→   891→    ELSE false
   892→   892→  END AS is_active,
   893→   893→  -- Dernier paiement
   894→   894→  (SELECT status FROM public.payments
   895→   895→   WHERE profile_id = p.id AND status = 'completed'
   896→   896→   ORDER BY paid_at DESC LIMIT 1) AS last_payment_status,
   897→   897→  (SELECT paid_at FROM public.payments
   898→   898→   WHERE profile_id = p.id AND status = 'completed'
   899→   899→   ORDER BY paid_at DESC LIMIT 1) AS last_payment_date,
   900→   900→  -- Usage ce mois
   901→   901→  COALESCE(u.reminders_sent, 0) AS reminders_used_this_month,
   902→   902→  COALESCE(u.appointments_created, 0) AS appointments_this_month
   903→   903→FROM public.profiles p
   904→   904→LEFT JOIN public.plans pl ON pl.id = p.plan
   905→   905→LEFT JOIN public.subscription_usage u
   906→   906→  ON u.profile_id = p.id AND u.usage_month = to_char(now(), 'YYYY-MM');
   907→   907→
   908→   908→-- Vue : revenus mensuels (pour le dashboard admin)
   909→   909→CREATE OR REPLACE VIEW public.v_monthly_revenue AS
   910→   910→SELECT
   911→   911→  to_char(paid_at, 'YYYY-MM') AS month,
   912→   912→  plan_id,
   913→   913→  plan_name,
   914→   914→  COUNT(*) AS payment_count,
   915→   915→  SUM(amount) AS total_amount,
   916→   916→  currency
   917→   917→FROM public.payments
   918→   918→WHERE status = 'completed' AND paid_at IS NOT NULL
   919→   919→GROUP BY to_char(paid_at, 'YYYY-MM'), plan_id, plan_name, currency
   920→   920→ORDER BY month DESC, total_amount DESC;
   921→   921→
   922→   922→-- Vue : abonnements expirant bientot (dans les 7 prochains jours)
   923→   923→CREATE OR REPLACE VIEW public.v_expiring_soon AS
   924→   924→SELECT
   925→   925→  s.id AS subscription_id,
   926→   926→  s.profile_id,
   927→   927→  s.plan_id,
   928→   928→  s.plan_name,
   929→   929→  s.ends_at,
   930→   930→  p.business_name,
   931→   931→  p.email,
   932→   932→  GREATEST(0, CEIL(EXTRACT(EPOCH FROM (s.ends_at - now())) / 86400)) AS days_left
   933→   933→FROM public.subscriptions s
   934→   934→JOIN public.profiles p ON p.id = s.profile_id
   935→   935→WHERE s.status = 'active'
   936→   936→  AND s.ends_at > now()
   937→   937→  AND s.ends_at < now() + INTERVAL '7 days'
   938→   938→ORDER BY s.ends_at ASC;
