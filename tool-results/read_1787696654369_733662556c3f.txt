   312→   312→-- 7. AJOUT COLONNES SUBSCRIPTION AU PROFIL
   313→   313→-- ============================================================
   314→   314→-- On ajoute les colonnes au profil existant pour un acces rapide
   315→   315→-- (evite un JOIN avec subscriptions a chaque requete).
   316→   316→-- Ces colonnes sont des COPIES miroirs de l'abonnement actif.
   317→   317→
   318→   318→-- Colonne plan (supprime l'ancienne si elle existe pour la recreer proprement)
   319→   319→DO $$ BEGIN
   320→   320→  IF EXISTS (
   321→   321→    SELECT 1 FROM information_schema.columns
   322→   322→    WHERE table_schema = 'public' AND table_name = 'profiles' AND column_name = 'plan'
   323→   323→  ) THEN
   324→   324→    ALTER TABLE public.profiles DROP COLUMN plan;
   325→   325→  END IF;
   326→   326→END $$;
   327→   327→
   328→   328→ALTER TABLE public.profiles
   329→   329→  ADD COLUMN IF NOT EXISTS plan TEXT DEFAULT 'free'
   330→   330→    CHECK (plan IN ('free', 'starter', 'pro', 'business')),
   331→   331→  ADD COLUMN IF NOT EXISTS subscription_status TEXT DEFAULT 'none'
   332→   332→    CHECK (subscription_status IN ('none', 'active', 'past_due', 'cancelled', 'expired')),
   333→   333→  ADD COLUMN IF NOT EXISTS subscription_start TIMESTAMPTZ,
   334→   334→  ADD COLUMN IF NOT EXISTS subscription_end TIMESTAMPTZ,
   335→   335→  ADD COLUMN IF NOT EXISTS subscription_id UUID REFERENCES public.subscriptions(id) ON DELETE SET NULL,
   336→   336→  ADD COLUMN IF NOT EXISTS chariow_customer_id TEXT;
   337→   337→
   338→   338→-- Renommer stripe_customer_id en chariow_customer_id si l'ancien existe
   339→   339→DO $$ BEGIN
   340→   340→  IF EXISTS (
   341→   341→    SELECT 1 FROM information_schema.columns
   342→   342→    WHERE table_schema = 'public' AND table_name = 'profiles'
   343→   343→      AND column_name = 'stripe_customer_id'
   344→   344→  ) THEN
   345→   345→    ALTER TABLE public.profiles RENAME COLUMN stripe_customer_id TO chariow_customer_id;
   346→   346→  END IF;
   347→   347→END $$;
   348→   348→
   349→   349→COMMENT ON COLUMN public.profiles.plan IS 'Plan actuel (miroir de la subscription active)';
   350→   350→COMMENT ON COLUMN public.profiles.subscription_status IS 'Statut de l\'abonnement actuel';
   351→   351→COMMENT ON COLUMN public.profiles.chariow_customer_id IS 'ID client chez Chariow (si applicable)';

   354→   354→-- 8. TRIGGER : Mise a jour miroir profil <-> subscription
   355→   355→-- ============================================================
   356→   356→
   357→   357→-- Quand une subscription est creee/activee, mettre a jour le profil
   358→   358→CREATE OR REPLACE FUNCTION public.sync_profile_subscription()
   359→   359→RETURNS TRIGGER AS $$
   360→   360→DECLARE
   361→   361→  v_plan_name TEXT;
   362→   362→  v_is_active BOOLEAN;
   363→   363→BEGIN
   364→   364→  -- Ne pas propager si on est deja en train de mettre a jour le profil
   365→   365→  -- (evite les boucles infinies avec le trigger ci-dessous)
   366→   366→  IF TG_OP = 'UPDATE' AND OLD.profile_id IS NOT DISTINCT FROM NEW.profile_id
   367→   367→     AND OLD.plan_id IS NOT DISTINCT FROM NEW.plan_id
   368→   368→     AND OLD.status IS NOT DISTINCT FROM NEW.status
   369→   369→     AND OLD.starts_at IS NOT DISTINCT FROM NEW.starts_at
   370→   370→     AND OLD.ends_at IS NOT DISTINCT FROM NEW.ends_at THEN
   371→   371→    RETURN NEW;
   372→   372→  END IF;
   373→   373→
   374→   374→  v_is_active := (NEW.status = 'active' AND NEW.ends_at > now());
   375→   375→
   376→   376→  IF v_is_active THEN
   377→   377→    v_plan_name := COALESCE(
   378→   378→      (SELECT name FROM public.plans WHERE id = NEW.plan_id),
   379→   379→      NEW.plan_name,
   380→   380→      NEW.plan_id
   381→   381→    );
   382→   382→
   383→   383→    UPDATE public.profiles SET
   384→   384→      plan            = NEW.plan_id,
   385→   385→      subscription_status = 'active',
   386→   386→      subscription_start   = NEW.starts_at,
   387→   387→      subscription_end     = NEW.ends_at,
   388→   388→      subscription_id      = NEW.id,
   389→   389→      updated_at           = now()
   390→   390→    WHERE id = NEW.profile_id;
   391→   391→  ELSE
   392→   392→    -- Si la subscription n'est plus active, retrograder le profil
   393→   393→    UPDATE public.profiles SET
   394→   394→      plan            = 'free',
   395→   395→      subscription_status = COALESCE(
   396→   396→        CASE NEW.status
   397→   397→          WHEN 'cancelled' THEN 'cancelled'
   398→   398→          WHEN 'expired'  THEN 'expired'
   399→   399→          WHEN 'upgraded' THEN 'active'  -- Le nouveau plan est deja actif
   400→   400→          WHEN 'downgraded' THEN 'active' -- Le nouveau plan est deja actif
   401→   401→          ELSE 'none'
   402→   402→        END,
   403→   403→        'none'
   404→   404→      ),
   405→   405→      updated_at           = now()
   406→   406→    WHERE id = NEW.profile_id
   407→   407→      AND subscription_id = NEW.id;  -- Ne mettre a jour que si c'est l'abonnement reference
   408→   408→  END IF;
   409→   409→
   410→   410→  RETURN NEW;
   411→   411→END;
   412→   412→$$ LANGUAGE plpgsql SECURITY DEFINER;
   413→   413→
   414→   414→DROP TRIGGER IF EXISTS trg_sync_profile_subscription ON public.subscriptions;
   415→   415→CREATE TRIGGER trg_sync_profile_subscription
   416→   416→  AFTER INSERT OR UPDATE OF status, ends_at, plan_id ON public.subscriptions
   417→   417→  FOR EACH ROW EXECUTE FUNCTION public.sync_profile_subscription();

   420→   420→-- 9. TRIGGER : Expiration automatique des abonnements
   421→   421→-- ============================================================
   422→   422→-- Fonction qui expire les abonnements dont la date de fin est passee.
   423→   423→-- A appeler via un cron job (GitHub Actions) ou manuellement.
   424→   424→
   425→   425→CREATE OR REPLACE FUNCTION public.expire_subscriptions()
   426→   426→RETURNS INTEGER AS $$
   427→   427→DECLARE
   428→   428→  v_count INTEGER;
   429→   429→BEGIN
   430→   430→  -- Verrouillage advisory pour eviter les executions concurrentes
   431→   431→  PERFORM pg_advisory_lock(54321);
   432→   432→
   433→   433→  UPDATE public.subscriptions
   434→   434→  SET status = 'expired',
   435→   435→      updated_at = now()
   436→   436→  WHERE status = 'active'

   479→   479→-- Incrementer un compteur d'usage mensuel
   480→   480→CREATE OR REPLACE FUNCTION public.increment_usage(
   481→   481→  p_profile_id UUID,
   482→   482→  p_counter TEXT   -- 'reminders_sent' ou 'appointments_created'
   483→   483→)
   484→   484→RETURNS INTEGER AS $$
   485→   485→DECLARE
   486→   486→  v_month TEXT := to_char(now(), 'YYYY-MM');
   487→   487→  v_new_count INTEGER;
   488→   488→BEGIN
   489→   489→  INSERT INTO public.subscription_usage (profile_id, usage_month, p_counter)
   490→   490→  VALUES (p_profile_id, v_month, 1)
   491→   491→  ON CONFLICT (profile_id, usage_month) DO UPDATE SET
   492→   492→    -- On ne peut pas utiliser un paramètre comme nom de colonne dynamiquement
   493→   493→    -- donc on fait un CASE
   494→   494→    reminders_sent = CASE
   495→   495→      WHEN p_counter = 'reminders_sent' THEN subscription_usage.reminders_sent + 1
   496→   496→      ELSE subscription_usage.reminders_sent
   497→   497→    END,
   498→   498→    appointments_created = CASE
   499→   499→      WHEN p_counter = 'appointments_created' THEN subscription_usage.appointments_created + 1
   500→   500→      ELSE subscription_usage.appointments_created
   501→   501→    END,
   502→   502→    updated_at = now()
   503→   503→  RETURNING
   504→   504→    CASE p_counter
   505→   505→      WHEN 'reminders_sent' THEN subscription_usage.reminders_sent
   506→   506→      ELSE subscription_usage.appointments_created
   507→   507→    END INTO v_new_count;
   508→   508→
   509→   509→  RETURN COALESCE(v_new_count, 0);
   510→   510→EXCEPTION WHEN OTHERS THEN
   511→   511→  RAISE WARNING 'increment_usage failed: %', SQLERRM;
   512→   512→  RETURN -1;
   513→   513→END;
   514→   514→$$ LANGUAGE plpgsql SECURITY DEFINER;

   516→   516→-- Verifier si un utilisateur peut utiliser une fonctionnalite
   517→   517→CREATE OR REPLACE FUNCTION public.can_use_feature(
   518→   518→  p_profile_id UUID,
   519→   519→  p_feature TEXT  -- 'professionals', 'services', 'reminders', 'appointments_per_day', 'clients'
   520→   520→)
   521→   521→RETURNS TABLE(allowed BOOLEAN, current_limit INTEGER, current_count INTEGER, plan_name TEXT) AS $$
   522→   522→DECLARE
   523→   523→  v_plan RECORD;
   524→   524→  v_current_count INTEGER := 0;
   525→   525→  v_limit INTEGER;
   526→   526→  v_month TEXT := to_char(now(), 'YYYY-MM');
   527→   527→BEGIN
   528→   528→  -- Recuperer le plan actuel du profil
   529→   529→  SELECT p.* INTO v_plan
   530→   530→  FROM public.plans p
   531→   531→  JOIN public.profiles pr ON pr.plan = p.id
   532→   532→  WHERE pr.id = p_profile_id;
   533→   533→
   534→   534→  IF NOT FOUND THEN
   535→   535→    -- Profil sans plan = free par defaut
   536→   536→    SELECT * INTO v_plan FROM public.plans WHERE id = 'free';
   537→   537→  END IF;
   538→   538→
   539→   539→  -- Calculer la limite selon la fonctionnalite
   540→   540→  v_limit := CASE p_feature
   541→   541→    WHEN 'professionals'        THEN v_plan.max_professionals
   542→   542→    WHEN 'services'             THEN v_plan.max_services
   543→   543→    WHEN 'reminders'            THEN v_plan.max_reminders_per_month
   544→   544→    WHEN 'appointments_per_day' THEN v_plan.max_appointments_per_day
   545→   545→    WHEN 'clients'              THEN v_plan.max_clients
   546→   546→    ELSE 0
   547→   547→  END;
   548→   548→
   549→   549→  -- Calculer le compteur actuel
   550→   550→  IF p_feature = 'reminders' THEN
   551→   551→    SELECT COALESCE(SUM(reminders_sent), 0) INTO v_current_count
   552→   552→    FROM public.subscription_usage
   553→   553→    WHERE profile_id = p_profile_id AND usage_month = v_month;
   554→   554→  ELSIF p_feature = 'appointments_per_day' THEN
   555→   555→    SELECT COUNT(*) INTO v_current_count
   556→   556→    FROM public.appointments
   557→   557→    WHERE profile_id = p_profile_id
   558→   558→      AND date(starts_at) = CURRENT_DATE
   559→   559→      AND status != 'cancelled';
   560→   560→  ELSIF p_feature = 'services' THEN
   561→   561→    SELECT COUNT(*) INTO v_current_count
   562→   562→    FROM public.services
   563→   563→    WHERE profile_id = p_profile_id AND is_active = true;
   564→   564→  ELSIF p_feature = 'clients' THEN
   565→   565→    SELECT COUNT(*) INTO v_current_count
   566→   566→    FROM public.clients
   567→   567→    WHERE profile_id = p_profile_id;
   568→   568→  ELSIF p_feature = 'professionals' THEN
   569→   569→    -- Pour l'instant, 1 profil = 1 professionnel
   570→   570→    v_current_count := 1;
   571→   571→  END IF;
   572→   572→
   573→   573→  -- -1 = illimite
   574→   574→  allowed := (v_limit = -1) OR (v_current_count < v_limit);
   575→   575→
   576→   576→  RETURN QUERY SELECT allowed, v_limit, v_current_count, v_plan.name;
   577→   577→  RETURN;
   578→   578→END;
   579→   579→$$ LANGUAGE plpgsql SECURITY DEFINER STABLE;

   581→   581→-- Creer un abonnement (appele par le webhook apres paiement valide)
   582→   582→CREATE OR REPLACE FUNCTION public.activate_subscription(
   583→   583→  p_profile_id UUID,
   584→   584→  p_plan_id    TEXT,
   585→   585→  p_payment_id UUID DEFAULT NULL,
   586→   586→  p_duration_days INTEGER DEFAULT 30
   587→   587→)
   588→   588→RETURNS UUID AS $$
   589→   589→DECLARE
   590→   590→  v_existing_active UUID;
   591→   591→  v_new_sub_id UUID;
   592→   592→  v_old_status TEXT;
   593→   593→BEGIN
   594→   594→  -- Verrouillage advisory par profil (evite les doubles activations)
   595→   595→  PERFORM pg_advisory_lock(hashtext(p_profile_id::text));
   596→   596→
   597→   597→  -- Trouver l'abonnement actuel s'il existe
   598→   598→  SELECT id INTO v_existing_active
   599→   599→  FROM public.subscriptions
   600→   600→  WHERE profile_id = p_profile_id AND status = 'active';
   601→   601→
   602→   602→  IF v_existing_active IS NOT NULL THEN
   603→   603→    -- Determiner si c'est un upgrade ou downgrade
   604→   604→    SELECT
   605→   605→      CASE
   606→   606→        WHEN (SELECT tier_priority FROM plans WHERE id = p_plan_id) >
   607→   607→             (SELECT tier_priority FROM plans WHERE id = s.plan_id)
   608→   608→        THEN 'upgraded'
   609→   609→        ELSE 'downgraded'
   610→   610→      END INTO v_old_status
   611→   611→    FROM public.subscriptions s WHERE id = v_existing_active;
   612→   612→
   613→   613→    -- Desactiver l'ancien abonnement
   614→   614→    UPDATE public.subscriptions
   615→   615→    SET status = v_old_status, updated_at = now()
   616→   616→    WHERE id = v_existing_active;
   617→   617→  END IF;
   618→   618→
   619→   619→  -- Creer le nouvel abonnement
   620→   620→  INSERT INTO public.subscriptions (
   621→   621→    profile_id, plan_id, plan_name,
   622→   622→    starts_at, ends_at,
   623→   623→    status, payment_id
   624→   624→  ) VALUES (
   625→   625→    p_profile_id, p_plan_id,
   626→   626→    COALESCE((SELECT name FROM plans WHERE id = p_plan_id), p_plan_id),
   627→   627→    now(),
   628→   628→    now() + (p_duration_days || ' days')::INTERVAL,
   629→   629→    'active',
   630→   630→    p_payment_id
   631→   631→  )
   632→   632→  RETURNING id INTO v_new_sub_id;
   633→   633→
   634→   634→  -- Le trigger trg_sync_profile_subscription va automatiquement
   635→   635→  -- mettre a jour les colonnes du profil
   636→   636→
   637→   637→  -- Release le verrou
   638→   638→  PERFORM pg_advisory_unlock(hashtext(p_profile_id::text));
   639→   639→
   640→   640→  RETURN v_new_sub_id;
   641→   641→END;
   642→   642→$$ LANGUAGE plpgsql SECURITY DEFINER;
   643→   643→
   644→   644→COMMENT ON FUNCTION public.activate_subscription() IS 'Active un abonnement apres paiement. Desactive l\'ancien si existant. Verrouille par profil pour eviter les doubles activations.';

   889→   889→-- Vue : resume abonnement d'un profil
   890→   890→CREATE OR REPLACE VIEW public.v_subscription_summary AS
   891→   891→SELECT
   892→   892→  p.id AS profile_id,
   893→   893→  p.plan,
   894→   894→  p.subscription_status,
   895→   895→  p.subscription_start,
   896→   896→  p.subscription_end,
   897→   897→  pl.name        AS plan_name,
   898→   898→  pl.price_monthly,
   899→   899→  pl.features,
   900→   900→  CASE
   901→   901→    WHEN p.subscription_status = 'active' AND p.subscription_end > now()
   902→   902→    THEN GREATEST(0, CEIL(EXTRACT(EPOCH FROM (p.subscription_end - now())) / 86400))
   903→   903→    ELSE 0
   904→   904→  END AS days_remaining,
   905→   905→  CASE
   906→   906→    WHEN p.subscription_status = 'active' AND p.subscription_end > now() THEN true
   907→   907→    ELSE false
   908→   908→  END AS is_active,
   909→   909→  -- Dernier paiement
   910→   910→  (SELECT status FROM public.payments
   911→   911→   WHERE profile_id = p.id AND status = 'completed'
   912→   912→   ORDER BY paid_at DESC LIMIT 1) AS last_payment_status,
   913→   913→  (SELECT paid_at FROM public.payments
   914→   914→   WHERE profile_id = p.id AND status = 'completed'
   915→   915→   ORDER BY paid_at DESC LIMIT 1) AS last_payment_date,
   916→   916→  -- Usage ce mois
   917→   917→  COALESCE(u.reminders_sent, 0) AS reminders_used_this_month,
   918→   918→  COALESCE(u.appointments_created, 0) AS appointments_this_month
   919→   919→FROM public.profiles p
   920→   920→LEFT JOIN public.plans pl ON pl.id = p.plan
   921→   921→LEFT JOIN public.subscription_usage u
   922→   922→  ON u.profile_id = p.id AND u.usage_month = to_char(now(), 'YYYY-MM');
   923→   923→
   924→   924→-- Vue : revenus mensuels (pour le dashboard admin)
   925→   925→CREATE OR REPLACE VIEW public.v_monthly_revenue AS
   926→   926→SELECT
   927→   927→  to_char(paid_at, 'YYYY-MM') AS month,
   928→   928→  plan_id,
   929→   929→  plan_name,
   930→   930→  COUNT(*) AS payment_count,
   931→   931→  SUM(amount) AS total_amount,
   932→   932→  currency
   933→   933→FROM public.payments
   934→   934→WHERE status = 'completed' AND paid_at IS NOT NULL
   935→   935→GROUP BY to_char(paid_at, 'YYYY-MM'), plan_id, plan_name, currency
   936→   936→ORDER BY month DESC, total_amount DESC;
   937→   937→
   938→   938→-- Vue : abonnements expirant bientot (dans les 7 prochains jours)
   939→   939→CREATE OR REPLACE VIEW public.v_expiring_soon AS
   940→   940→SELECT
   941→   941→  s.id AS subscription_id,
   942→   942→  s.profile_id,
   943→   943→  s.plan_id,
   944→   944→  s.plan_name,
   945→   945→  s.ends_at,
   946→   946→  p.business_name,
   947→   947→  p.email,
   948→   948→  GREATEST(0, CEIL(EXTRACT(EPOCH FROM (s.ends_at - now())) / 86400)) AS days_left
   949→   949→FROM public.subscriptions s
   950→   950→JOIN public.profiles p ON p.id = s.profile_id
   951→   951→WHERE s.status = 'active'
   952→   952→  AND s.ends_at > now()
   953→   953→  AND s.ends_at < now() + INTERVAL '7 days'
   954→   954→ORDER BY s.ends_at ASC;
