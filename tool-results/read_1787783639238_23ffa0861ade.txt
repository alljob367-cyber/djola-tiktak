     1→'use client';
     2→
     3→import { use, useState, useEffect, useCallback, useMemo } from 'react';
     4→import { useRouter, useSearchParams } from 'next/navigation';
     5→import { motion, AnimatePresence } from 'framer-motion';
     6→import { toast } from 'sonner';
     7→import { z } from 'zod';
     8→import { Button } from '@/components/ui/button';
     9→import { Input } from '@/components/ui/input';
    10→import { Label } from '@/components/ui/label';
    11→import { Card, CardContent } from '@/components/ui/card';
    12→import { Skeleton } from '@/components/ui/skeleton';
    13→import {
    14→  Clock,
    15→  ChevronLeft,
    16→  Check,
    17→  Share2,
    18→  ArrowLeft,
    19→  CalendarDays,
    20→  User,
    21→  CreditCard,
    22→  Sparkles,
    23→  AlertCircle,
    24→} from 'lucide-react';
    25→import {
    26→  formatCurrency,
    27→  DAY_NAMES_SHORT_FR,
    28→  MONTH_NAMES_FR,
    29→} from '@/lib/availability/engine';
    30→
    31→// ------------------------------------------------------------------
    32→// Types
    33→// ------------------------------------------------------------------
    34→
    35→interface Service {
    36→  id: string;
    37→  name: string;
    38→  description: string;
    39→  price: number;
    40→  duration_minutes: number;
    41→}
    42→
    43→interface Profile {
    44→  id: string;
    45→  business_name: string;
    46→  slug: string;
    47→  phone: string;
    48→  avatar_url: string;
    49→  currency: string;
    50→}
    51→
    52→interface Slot {
    53→  starts_at: string;
    54→  ends_at: string;
    55→}
    56→
    57→interface BookingPageProps {
    58→  params: Promise<{ slug: string }>;
    59→  searchParams: Promise<{ service?: string }>;
    60→}
    61→
    62→// ------------------------------------------------------------------
    63→// Schema
    64→// ------------------------------------------------------------------
    65→
    66→const clientInfoSchema = z.object({
    67→  client_name: z.string().min(1, 'Le nom est requis'),
    68→  client_phone: z.string().min(1, 'Le téléphone est requis'),
    69→  client_email: z.string().email('Email invalide').optional().or(z.literal('')),
    70→});
    71→
    72→type ClientInfo = z.infer<typeof clientInfoSchema>;
    73→
    74→// ------------------------------------------------------------------
    75→// Steps definition
    76→// ------------------------------------------------------------------
    77→
    78→const STEPS = [
    79→  { id: 'service', label: 'Service', icon: Sparkles },
    80→  { id: 'date', label: 'Date', icon: CalendarDays },
    81→  { id: 'time', label: 'Horaire', icon: Clock },
    82→  { id: 'info', label: 'Coordonnées', icon: User },
    83→  { id: 'confirm', label: 'Confirmation', icon: Check },
    84→] as const;
    85→
    86→type StepId = (typeof STEPS)[number]['id'];
    87→
    88→// ------------------------------------------------------------------
    89→// Animation helpers
    90→// ------------------------------------------------------------------
    91→
    92→const slideVariants = {
    93→  enter: (direction: number) => ({
    94→    x: direction > 0 ? 200 : -200,
    95→    opacity: 0,
    96→  }),
    97→  center: {
    98→    x: 0,
    99→    opacity: 1,
   100→  },
   101→  exit: (direction: number) => ({
   102→    x: direction < 0 ? 200 : -200,
   103→    opacity: 0,
   104→  }),
   105→};
   106→
   107→// ------------------------------------------------------------------
   108→// Component
   109→// ------------------------------------------------------------------
   110→
   111→export default function BookingPage({ params, searchParams }: BookingPageProps) {
   112→  const { slug } = use(params);
   113→  const sp = use(searchParams);
   114→  const router = useRouter();
   115→
   116→  // ---- data state ----
   117→  const [profile, setProfile] = useState<Profile | null>(null);
   118→  const [services, setServices] = useState<Service[]>([]);
   119→  const [loading, setLoading] = useState(true);
   120→
   121→  // ---- step state ----
   122→  const [currentStep, setCurrentStep] = useState(0);
   123→  const [direction, setDirection] = useState(1);
   124→
   125→  // ---- selections ----
   126→  const [selectedService, setSelectedService] = useState<Service | null>(null);
   127→  const [selectedDate, setSelectedDate] = useState<Date | null>(null);
   128→  const [selectedSlot, setSelectedSlot] = useState<Slot | null>(null);
   129→  const [clientInfo, setClientInfo] = useState<ClientInfo>({
   130→    client_name: '',
   131→    client_phone: '',
   132→    client_email: '',
   133→  });
   134→  const [clientErrors, setClientErrors] = useState<
   135→    Partial<Record<keyof ClientInfo, string>>
   136→  >({});
   137→
   138→  // ---- async states ----
   139→  const [slotsLoading, setSlotsLoading] = useState(false);
   140→  const [slots, setSlots] = useState<Slot[]>([]);
   141→  const [submitting, setSubmitting] = useState(false);
   142→  const [booked, setBooked] = useState(false);
   143→
   144→  // ---- Fetch profile & services ----
   145→  useEffect(() => {
   146→    async function fetchProfile() {
   147→      try {
   148→        const res = await fetch(`/api/profiles/${slug}?slug=${slug}`);
   149→        if (!res.ok) {
   150→          router.push(`/${slug}`);
   151→          return;
   152→        }
   153→        const json = await res.json();
   154→        const data = json.data;
   155→        setProfile({
   156→          id: data.id,
   157→          business_name: data.business_name,
   158→          slug: data.slug,
   159→          phone: data.phone,
   160→          avatar_url: data.avatar_url,
   161→          currency: data.currency,
   162→        });
   163→        const svc: Service[] = (data.services || []).map(
   164→          (s: Record<string, unknown>) => ({
   165→            id: s.id as string,
   166→            name: s.name as string,
   167→            description: (s.description as string) || '',
   168→            price: s.price as number,
   169→            duration_minutes: s.duration_minutes as number,
   170→          })
   171→        );
   172→        setServices(svc);
   173→
   174→        // Pre-select service if in URL
   175→        if (sp.service) {
   176→          const found = svc.find((s) => s.id === sp.service);
   177→          if (found) {
   178→            setSelectedService(found);
   179→            setCurrentStep(1); // Jump to date step
   180→          }
   181→        }
   182→      } catch {
   183→        toast.error('Impossible de charger les informations');
   184→      } finally {
   185→        setLoading(false);
   186→      }
   187→    }
   188→    fetchProfile();
   189→    }, [slug]);
   190→
   191→  // ---- Next 14 days ----
   192→  const next14Days = useMemo(() => {
   193→    const days: Date[] = [];
   194→    const today = new Date();
   195→    for (let i = 1; i <= 14; i++) {
   196→      const d = new Date(today);
   197→      d.setDate(today.getDate() + i);
   198→      days.push(d);
   199→    }
   200→    return days;
   201→  }, []);
   202→
   203→  // ---- Fetch slots when date is selected ----
   204→  const fetchSlots = useCallback(
   205→    async (date: Date) => {
   206→      if (!selectedService) return;
   207→      setSlotsLoading(true);
   208→      setSlots([]);
   209→      setSelectedSlot(null);
   210→      try {
   211→        const dateStr = date.toISOString().split('T')[0];
   212→        const res = await fetch(
   213→          `/api/bookings/availability?slug=${slug}&service_id=${selectedService.id}&date=${dateStr}`
   214→        );
   215→        if (!res.ok) {
   216→          toast.error('Erreur lors du chargement des créneaux');
   217→          return;
   218→        }
   219→        const json = await res.json();
   220→        const data = json.data;
   221→        setSlots(Array.isArray(data) ? data : data?.slots || []);
   222→      } catch {
   223→        toast.error('Erreur réseau');
   224→      } finally {
   225→        setSlotsLoading(false);
   226→      }
   227→    },
   228→    [slug, selectedService]
   229→  );
   230→
   231→  // ---- Navigation helpers ----
   232→  const goNext = () => {
   233→    setDirection(1);
   234→    setCurrentStep((s) => Math.min(s + 1, STEPS.length - 1));
   235→  };
   236→
   237→  const goBack = () => {
   238→    setDirection(-1);
   239→    setCurrentStep((s) => Math.max(s - 1, 0));
   240→  };
   241→
   242→  const canGoNext = (): boolean => {
   243→    switch (STEPS[currentStep].id) {
   244→      case 'service':
   245→        return selectedService !== null;
   246→      case 'date':
   247→        return selectedDate !== null;
   248→      case 'time':
   249→        return selectedSlot !== null;
   250→      case 'info': {
   251→        const result = clientInfoSchema.safeParse(clientInfo);
   252→        return result.success;
   253→      }
   254→      case 'confirm':
   255→        return false;
   256→      default:
   257→        return false;
   258→    }
   259→  };
   260→
   261→  const handleNext = () => {
   262→    if (!canGoNext()) return;
   263→
   264→    // Validate info step
   265→    if (STEPS[currentStep].id === 'info') {
   266→      const result = clientInfoSchema.safeParse(clientInfo);
   267→      if (!result.success) {
   268→        const errors: Partial<Record<keyof ClientInfo, string>> = {};
   269→        for (const issue of result.error.issues) {
   270→          const key = issue.path[0] as keyof ClientInfo;
   271→          errors[key] = issue.message;
   272→        }
   273→        setClientErrors(errors);
   274→        return;
   275→      }
   276→      setClientErrors({});
   277→    }
   278→
   279→    // Fetch slots when moving from date to time
   280→    if (STEPS[currentStep].id === 'date' && selectedDate) {
   281→      fetchSlots(selectedDate);
   282→    }
   283→
   284→    goNext();
   285→  };
   286→
   287→  const handleDateSelect = (date: Date) => {
   288→    setSelectedDate(date);
   289→    setSelectedSlot(null);
   290→  };
   291→
   292→  const handleConfirm = async () => {
   293→    if (!selectedService || !selectedSlot) return;
   294→    setSubmitting(true);
   295→    try {
   296→      const res = await fetch('/api/bookings/public', {
   297→        method: 'POST',
   298→        headers: { 'Content-Type': 'application/json' },
   299→        body: JSON.stringify({
   300→          service_id: selectedService.id,
   301→          client_name: clientInfo.client_name,
   302→          client_phone: clientInfo.client_phone,
   303→          client_email: clientInfo.client_email || '',
   304→          starts_at: selectedSlot.starts_at,
   305→        }),
   306→      });
   307→
   308→      if (!res.ok) {
   309→        const json = await res.json().catch(() => ({}));
   310→        toast.error(json.error || 'Erreur lors de la réservation');
   311→        return;
   312→      }
   313→
   314→      setBooked(true);
   315→      goNext();
   316→    } catch {
   317→      toast.error('Erreur réseau, veuillez réessayer');
   318→    } finally {
   319→      setSubmitting(false);
   320→    }
   321→  };
   322→
   323→  const handleShare = async () => {
   324→    if (navigator.share) {
   325→      try {
   326→        await navigator.share({
   327→          title: 'Rendez-vous confirmé',
   328→          text: `Rendez-vous pour ${selectedService?.name} le ${formatDateFR(selectedDate!)} à ${formatTimeFR(selectedSlot!.starts_at)}`,
   329→        });
   330→      } catch {
   331→        // user cancelled
   332→      }
   333→    } else {
   334→      await navigator.clipboard.writeText(
   335→        `Rendez-vous pour ${selectedService?.name} le ${formatDateFR(selectedDate!)} à ${formatTimeFR(selectedSlot!.starts_at)}`
   336→      );
   337→      toast.success('Détails copiés dans le presse-papier');
   338→    }
   339→  };
   340→
   341→  // ---- Formatting helpers ----
   342→  function formatDateFR(date: Date): string {
   343→    return `${date.getDate()} ${MONTH_NAMES_FR[date.getMonth()]} ${date.getFullYear()}`;
   344→  }
   345→
   346→  function formatTimeFR(iso: string): string {
   347→    const d = new Date(iso);
   348→    return d.toLocaleTimeString('fr-FR', { hour: '2-digit', minute: '2-digit' });
   349→  }
   350→
   351→  // ---- Render helpers ----
   352→
   353→  const renderStepper = () => {
   354→    if (booked) return null;
   355→    return (
   356→      <div className="mb-6 flex items-center justify-center gap-0 px-2">
   357→        {STEPS.map((step, i) => {
   358→          const isActive = i === currentStep;
   359→          const isDone = i < currentStep;
   360→          const StepIcon = step.icon;
   361→          return (
   362→            <div key={step.id} className="flex items-center">
   363→              <div className="flex flex-col items-center">
   364→                <div
   365→                  className={`flex size-10 items-center justify-center rounded-full border-2 transition-colors sm:size-11 ${
   366→                    isActive
   367→                      ? 'border-emerald-600 bg-emerald-600 text-white'
   368→                      : isDone
   369→                        ? 'border-emerald-600 bg-emerald-100 text-emerald-700'
   370→                        : 'border-muted-foreground/25 bg-background text-muted-foreground'
   371→                  }`}
   372→                >
   373→                  {isDone ? (
   374→                    <Check className="size-5" strokeWidth={3} />
   375→                  ) : (
   376→                    <StepIcon className="size-4 sm:size-5" />
   377→                  )}
   378→                </div>
   379→                <span
   380→                  className={`mt-1 text-[10px] font-medium sm:text-xs ${
   381→                    isActive ? 'text-emerald-700' : 'text-muted-foreground'
   382→                  }`}
   383→                >
   384→                  {step.label}
   385→                </span>
   386→              </div>
   387→              {i < STEPS.length - 1 && (
   388→                <div
   389→                  className={`mx-1.5 h-0.5 w-6 sm:mx-2 sm:w-10 ${
   390→                    i < currentStep ? 'bg-emerald-500' : 'bg-muted-foreground/20'
   391→                  }`}
   392→                />
   393→              )}
   394→            </div>
   395→          );
   396→        })}
   397→      </div>
   398→    );
   399→  };
   400→
   401→  // ---- Step 1: Select Service ----
   402→  const renderServiceStep = () => (
   403→    <div className="space-y-3">
   404→      <h2 className="text-lg font-semibold text-foreground">
   405→        Choisissez un service
   406→      </h2>
   407→      {services.map((svc) => {
   408→        const isSelected = selectedService?.id === svc.id;
   409→        return (
   410→          <button
   411→            key={svc.id}
   412→            type="button"
   413→            onClick={() => setSelectedService(svc)}
   414→            className={`w-full min-h-[44px] rounded-xl border-2 p-4 text-left transition-all ${
   415→              isSelected
   416→                ? 'border-emerald-600 bg-emerald-50 shadow-sm'
   417→                : 'border-border bg-card hover:border-emerald-300 hover:bg-emerald-50/50'
   418→            }`}
   419→          >
   420→            <div className="flex items-center justify-between gap-3">
   421→              <div className="min-w-0 flex-1">
   422→                <p className="truncate font-semibold text-foreground">
   423→                  {svc.name}
   424→                </p>
   425→                {svc.description && (
   426→                  <p className="mt-0.5 line-clamp-1 text-sm text-muted-foreground">
   427→                    {svc.description}
   428→                  </p>
   429→                )}
   430→                <div className="mt-1.5 flex items-center gap-3 text-sm">
   431→                  <span className="inline-flex items-center gap-1 text-muted-foreground">
   432→                    <Clock className="size-3.5" />
   433→                    {svc.duration_minutes} min
   434→                  </span>
   435→                  <span className="font-bold text-emerald-700">
   436→                    {formatCurrency(svc.price, profile?.currency)}
   437→                  </span>
   438→                </div>
   439→              </div>
   440→              <div
   441→                className={`flex size-6 shrink-0 items-center justify-center rounded-full border-2 transition-colors ${
   442→                  isSelected
   443→                    ? 'border-emerald-600 bg-emerald-600 text-white'
   444→                    : 'border-muted-foreground/30'
   445→                }`}
   446→              >
   447→                {isSelected && <Check className="size-3.5" strokeWidth={3} />}
   448→              </div>
   449→            </div>
   450→          </button>
   451→        );
   452→      })}
   453→    </div>
   454→  );
   455→
   456→  // ---- Step 2: Select Date ----
   457→  const renderDateStep = () => (
   458→    <div className="space-y-4">
   459→      <h2 className="text-lg font-semibold text-foreground">
   460→        Choisissez une date
   461→      </h2>
   462→      <div className="grid grid-cols-3 gap-2 sm:grid-cols-4 sm:gap-3">
   463→        {next14Days.map((day) => {
   464→          const dayName = DAY_NAMES_SHORT_FR[day.getDay()];
   465→          const monthName = MONTH_NAMES_FR[day.getMonth()];
   466→          const dayNum = day.getDate();
   467→          const isSelected =
   468→            selectedDate &&
   469→            selectedDate.getDate() === day.getDate() &&
   470→            selectedDate.getMonth() === day.getMonth();
   471→          return (
   472→            <button
   473→              key={day.toISOString()}
   474→              type="button"
   475→              onClick={() => handleDateSelect(day)}
   476→              className={`flex min-h-[72px] flex-col items-center justify-center rounded-xl border-2 px-2 py-3 transition-all ${
   477→                isSelected
   478→                  ? 'border-emerald-600 bg-emerald-600 text-white shadow-md'
   479→                  : 'border-border bg-card hover:border-emerald-300 hover:bg-emerald-50/50'
   480→              }`}
   481→            >
   482→              <span
   483→                className={`text-xs font-medium ${isSelected ? 'text-emerald-100' : 'text-muted-foreground'}`}
   484→              >
   485→                {dayName}
   486→              </span>
   487→              <span className="text-xl font-bold leading-tight">{dayNum}</span>
   488→              <span
   489→                className={`text-[10px] ${isSelected ? 'text-emerald-100' : 'text-muted-foreground'}`}
   490→              >
   491→                {monthName}
   492→              </span>
   493→            </button>
   494→          );
   495→        })}
   496→      </div>
   497→    </div>
   498→  );
   499→
   500→  // ---- Step 3: Select Time ----
   501→  const renderTimeStep = () => (
   502→    <div className="space-y-4">
   503→      <h2 className="text-lg font-semibold text-foreground">
   504→        Choisissez un créneau
   505→      </h2>
   506→      <p className="text-sm text-muted-foreground">
   507→        {selectedDate && formatDateFR(selectedDate)}
   508→      </p>
   509→      {slotsLoading ? (
   510→        <div className="grid grid-cols-3 gap-2">
   511→          {Array.from({ length: 6 }).map((_, i) => (
   512→            <Skeleton key={i} className="h-12 rounded-xl" />
   513→          ))}
   514→        </div>
   515→      ) : slots.length === 0 ? (
   516→        <div className="flex flex-col items-center gap-3 rounded-xl border border-dashed border-muted-foreground/30 py-12 text-center">
   517→          <AlertCircle className="size-10 text-muted-foreground/50" />
   518→          <p className="text-sm font-medium text-muted-foreground">
   519→            Aucun créneau disponible
   520→          </p>
   521→          <p className="text-xs text-muted-foreground/70">
   522→            Essayez une autre date
   523→          </p>
   524→        </div>
   525→      ) : (
   526→        <div className="grid grid-cols-3 gap-2 sm:grid-cols-4">
   527→          {slots.map((slot, i) => {
   528→            const isSelected = selectedSlot?.starts_at === slot.starts_at;
   529→            return (
   530→              <button
   531→                key={i}
   532→                type="button"
   533→                onClick={() => setSelectedSlot(slot)}
   534→                className={`min-h-[48px] rounded-xl border-2 px-3 py-2.5 text-center text-sm font-semibold transition-all ${
   535→                  isSelected
   536→                    ? 'border-emerald-600 bg-emerald-600 text-white shadow-md'
   537→                    : 'border-border bg-card hover:border-emerald-300 hover:bg-emerald-50/50'
   538→                }`}
   539→              >
   540→                {formatTimeFR(slot.starts_at)}
   541→              </button>
   542→            );
   543→          })}
   544→        </div>
   545→      )}
   546→    </div>
   547→  );
   548→
   549→  // ---- Step 4: Client Info ----
   550→  const renderInfoStep = () => (
   551→    <div className="space-y-5">
   552→      <h2 className="text-lg font-semibold text-foreground">
   553→        Vos coordonnées
   554→      </h2>
   555→      <div className="space-y-4">
   556→        <div className="space-y-2">
   557→          <Label htmlFor="client_name" className="text-sm">
   558→            Nom complet <span className="text-destructive">*</span>
   559→          </Label>
   560→          <Input
   561→            id="client_name"
   562→            type="text"
   563→            placeholder="Jean Dupont"
   564→            value={clientInfo.client_name}
   565→            onChange={(e) => {
   566→              setClientInfo((prev) => ({ ...prev, client_name: e.target.value }));
   567→              if (clientErrors.client_name) {
   568→                setClientErrors((prev) => ({ ...prev, client_name: undefined }));
   569→              }
   570→            }}
   571→            className={`min-h-[44px] ${clientErrors.client_name ? 'border-destructive' : ''}`}
   572→            autoComplete="name"
   573→          />
   574→          {clientErrors.client_name && (
   575→            <p className="text-xs text-destructive">{clientErrors.client_name}</p>
   576→          )}
   577→        </div>
   578→
   579→        <div className="space-y-2">
   580→          <Label htmlFor="client_phone" className="text-sm">
   581→            Téléphone <span className="text-destructive">*</span>
   582→          </Label>
   583→          <Input
   584→            id="client_phone"
   585→            type="tel"
   586→            placeholder="+237 6XX XXX XXX"
   587→            value={clientInfo.client_phone}
   588→            onChange={(e) => {
   589→              setClientInfo((prev) => ({ ...prev, client_phone: e.target.value }));
   590→              if (clientErrors.client_phone) {
   591→                setClientErrors((prev) => ({ ...prev, client_phone: undefined }));
   592→              }
   593→            }}
   594→            className={`min-h-[44px] ${clientErrors.client_phone ? 'border-destructive' : ''}`}
   595→            autoComplete="tel"
   596→          />
   597→          {clientErrors.client_phone && (
   598→            <p className="text-xs text-destructive">{clientErrors.client_phone}</p>
   599→          )}
   600→        </div>
   601→
   602→        <div className="space-y-2">
   603→          <Label htmlFor="client_email" className="text-sm">
   604→            Email <span className="text-muted-foreground">(facultatif)</span>
   605→          </Label>
   606→          <Input
   607→            id="client_email"
