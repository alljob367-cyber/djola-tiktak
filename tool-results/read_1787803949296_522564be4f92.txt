    30→import {

    37→} from '@/components/ui/dialog';
    38→import {

    45→  AlertDialogHeader,
    46→  AlertDialogTitle,
    47→  AlertDialogTrigger,
    48→} from '@/components/ui/alert-dialog';
    49→import { Input } from '@/components/ui/input';

    55→
    56→// ── Animation ────────────────────────────────────────────────
    57→const containerVariants = {
    58→  hidden: { opacity: 0 },
    59→  visible: {
    60→    opacity: 1,

    62→  },
    63→};
    64→
    65→const itemVariants = {
    66→  hidden: { opacity: 0, y: 16 },

    68→};
    69→
    70→// ── Component ────────────────────────────────────────────────
    71→export default function SettingsPage() {
    72→  const router = useRouter();
    73→  const supabase = createClient();
    74→
    75→  const [email, setEmail] = useState('');
    76→  const [loadingEmail, setLoadingEmail] = useState(true);
    77→
    78→  // Password dialog
    79→  const [pwdDialogOpen, setPwdDialogOpen] = useState(false);
    80→  const [newPassword, setNewPassword] = useState('');
    81→  const [confirmPassword, setConfirmPassword] = useState('');
    82→  const [pwdError, setPwdError] = useState('');
    83→  const [pwdSubmitting, setPwdSubmitting] = useState(false);
    84→
    85→  // Danger actions
    86→  const [deactivating, setDeactivating] = useState(false);
    87→  const [deleting, setDeleting] = useState(false);
    88→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false);
    89→  const [deleteConfirmText, setDeleteConfirmText] = useState('');
    90→
    91→  // Payment methods state
    92→  const [pmEnabled, setPmEnabled] = useState(false);
    93→  const [omPhone, setOmPhone] = useState('');
    94→  const [omName, setOmName] = useState('');
    95→  const [mtnPhone, setMtnPhone] = useState('');
    96→  const [mtnName, setMtnName] = useState('');
    97→  const [pmInstructions, setPmInstructions] = useState('');
    98→  const [pmLoading, setPmLoading] = useState(true);
    99→  const [pmSaving, setPmSaving] = useState(false);
   100→
   101→  // ── Fetch user email & payment settings ─────────────────
   102→  useEffect(() => {
   103→    (async () => {
   104→      try {
   105→        const {
   106→          data: { user },
   107→        } = await supabase.auth.getUser();
   108→        if (user) setEmail(user.email ?? '');
   109→
   110→        // Fetch profile for payment settings
   111→        const { data: profile } = await supabase
   112→          .from('profiles')
   113→          .select('payment_methods_enabled, orange_money_phone, orange_money_name, mtn_momo_phone, mtn_momo_name, payment_instructions')
   114→          .single();
   115→
   116→        if (profile) {
   117→          setPmEnabled(profile.payment_methods_enabled ?? false);
   118→          setOmPhone(profile.orange_money_phone ?? '');
   119→          setOmName(profile.orange_money_name ?? '');
   120→          setMtnPhone(profile.mtn_momo_phone ?? '');
   121→          setMtnName(profile.mtn_momo_name ?? '');
   122→          setPmInstructions(profile.payment_instructions ?? '');
   123→        }
   124→      } catch {
   125→        // silent
   126→      } finally {
   127→        setLoadingEmail(false);
   128→        setPmLoading(false);
   129→      }
   130→    })();
   131→  }, [supabase]);
   132→
   133→  // ── Save payment methods ────────────────────────────────
   134→  const handleSavePaymentMethods = async () => {
   135→    setPmSaving(true);
   136→    try {
   137→      const { error } = await supabase
   138→        .from('profiles')
   139→        .update({
   140→          payment_methods_enabled: pmEnabled,
   141→          orange_money_phone: omPhone || null,
   142→          orange_money_name: omName || null,
   143→          mtn_momo_phone: mtnPhone || null,
   144→          mtn_momo_name: mtnName || null,
   145→          payment_instructions: pmInstructions || null,
   146→          updated_at: new Date().toISOString(),
   147→        });
   148→
   149→      if (error) throw error;
   150→      toast.success('Moyens de paiement enregistrés');
   151→    } catch {
   152→      toast.error('Erreur lors de l\'enregistrement');
   153→    } finally {
   154→      setPmSaving(false);
   155→    }
   156→  };
   157→
   158→  // ── Copy to clipboard ──────────────────────────────────
   159→  const [copied, setCopied] = useState(false);
   160→  const handleCopy = (text: string) => {
   161→    navigator.clipboard.writeText(text).then(() => {
   162→      setCopied(true);
   163→      toast.success('Copié !');
   164→      setTimeout(() => setCopied(false), 2000);
   165→    });
   166→  };
   167→
   168→  // ── Change password ───────────────────────────────────
   169→  const handleChangePassword = async () => {
   170→    setPwdError('');
   171→
   172→    if (!newPassword) {
   173→      setPwdError('Le nouveau mot de passe est requis');
   174→      return;
   175→    }
   176→    if (newPassword.length < 6) {
   177→      setPwdError('Le mot de passe doit contenir au moins 6 caractères');
   178→      return;
   179→    }
   180→    if (newPassword !== confirmPassword) {
   181→      setPwdError('Les mots de passe ne correspondent pas');
   182→      return;
   183→    }
   184→
   185→    setPwdSubmitting(true);
   186→    try {
   187→      const { error } = await supabase.auth.updateUser({
   188→        password: newPassword,
   189→      });
   190→
   191→      if (error) throw error;
   192→
   193→      toast.success('Mot de passe modifié avec succès');
   194→      setPwdDialogOpen(false);
   195→      setNewPassword('');
   196→      setConfirmPassword('');
   197→    } catch (err: unknown) {
   198→      const msg =
   199→        err instanceof Error ? err.message : 'Erreur lors de la modification';
   200→      setPwdError(msg);
   201→    } finally {
   202→      setPwdSubmitting(false);
   203→    }
   204→  };
   205→
   206→  // ── Logout ────────────────────────────────────────────
   207→  const handleLogout = async () => {
   208→    try {
   209→      await supabase.auth.signOut();
   210→      router.push('/login');
   211→      router.refresh();
   212→    } catch {
   213→      toast.error('Erreur lors de la déconnexion');
   214→    }
   215→  };
   216→
   217→  // ── Deactivate account ────────────────────────────────
   218→  const handleDeactivate = async () => {
   219→    setDeactivating(true);
   220→    try {
   221→      const { data: { user } } = await supabase.auth.getUser();
   222→      if (!user) throw new Error('Non autorisé');
   223→
   224→      const res = await fetch('/api/profiles', {
   225→        method: 'PUT',
   226→        headers: { 'Content-Type': 'application/json' },
   227→        body: JSON.stringify({
   228→          business_name: '',
   229→          slug: '',
   230→          description: '',
   231→          phone: '',
   232→          email: '',
   233→          currency: 'XAF',
   234→          timezone: 'Africa/Malabo',
   235→          is_active: false,
   236→        }),
   237→      });
   238→
   239→      toast.info('Compte désactivé');
   240→      await supabase.auth.signOut();
   241→      router.push('/login');
   242→    } catch {
   243→      toast.error('Erreur lors de la désactivation');
   244→    } finally {
   245→      setDeactivating(false);
   246→    }
   247→  };
   248→
   249→  // ── Delete account ────────────────────────────────────
   250→  const handleDeleteAccount = async () => {
   251→    if (deleteConfirmText !== 'SUPPRIMER') {
   252→      toast.error('Veuillez taper SUPPRIMER pour confirmer');
   253→      return;
   254→    }
   255→
   256→    setDeleting(true);
   257→    try {
   258→      await supabase.auth.signOut();
   259→      toast.success('Compte supprimé. Vous pouvez maintenant contacter le support pour supprimer définitivement vos données.');
   260→      router.push('/login');
   261→    } catch {
   262→      toast.error('Erreur lors de la suppression');
   263→    } finally {
   264→      setDeleting(false);
   265→    }
   266→  };
   267→
   268→  // ── Render ─────────────────────────────────────────────
   269→  return (
   270→    <motion.div
   271→      variants={containerVariants}
   272→      initial="hidden"
   273→      animate="visible"
   274→      className="space-y-6 max-w-2xl"
   275→    >
   276→      {/* Header */}
   277→      <motion.div variants={itemVariants}>
   278→        <h1 className="text-2xl font-bold tracking-tight">Paramètres</h1>
   279→        <p className="text-sm text-muted-foreground mt-1">
   280→          Gérez votre compte et vos préférences
   281→        </p>
   282→      </motion.div>
   283→
   284→      {/* ── Payment Methods Section ──────────────────────────── */}
   285→      <motion.div variants={itemVariants}>
   286→        <Card>
   287→          <CardHeader>
   288→            <CardTitle className="text-base flex items-center gap-2">
   289→              <Wallet size={16} className="text-emerald-600" />
   290→              Moyens de paiement locaux
   291→            </CardTitle>
   292→            <CardDescription>
   293→              Configurez vos numéros pour recevoir des paiements anticipés de vos clients. Les clients qui paient en avance seront en priorité.
   294→            </CardDescription>
   295→          </CardHeader>
   296→          <CardContent className="space-y-6">
   297→            {/* Toggle */}
   298→            <div className="flex items-center justify-between gap-4">
   299→              <div className="min-w-0">
   300→                <p className="text-sm font-medium">Activer les paiements anticipés</p>
   301→                <p className="text-xs text-muted-foreground">
   302→                  Vos clients pourront vous payer en avance pour être prioritaires
   303→                </p>
   304→              </div>
   305→              <Switch
   306→                checked={pmEnabled}
   307→                onCheckedChange={setPmEnabled}
   308→                disabled={pmSaving}
   309→              />
   310→            </div>
   311→
   312→            {pmEnabled && (
   313→              <>
   314→                <Separator />
   315→
   316→                {/* Info banner */}
   317→                <div className="rounded-lg border border-blue-200 bg-blue-50 p-3 dark:border-blue-900/50 dark:bg-blue-950/20">
   318→                  <div className="flex gap-2">
   319→                    <Info size={16} className="mt-0.5 shrink-0 text-blue-600 dark:text-blue-400" />
   320→                    <div className="text-xs text-blue-700 dark:text-blue-300">
   321→                      <p className="font-medium">Comment ça marche ?</p>
   322→                      <p className="mt-1">Vos clients verront vos numéros de paiement sur votre page de réservation. Ils pourront vous envoyer le montant du service en avance via Orange Money ou MTN MoMo. Les réservations payées en avance seront marquées comme prioritaires.</p>
   323→                    </div>
   324→                  </div>
   325→                </div>
   326→
   327→                {/* Orange Money */}
   328→                <div className="space-y-3">
   329→                  <div className="flex items-center gap-2">
   330→                    <div className="flex size-8 items-center justify-center rounded-lg bg-orange-100 dark:bg-orange-900/40">
   331→                      <span className="text-sm font-bold text-orange-600 dark:text-orange-400">OM</span>
   332→                    </div>
   333→                    <div>
   334→                      <p className="text-sm font-medium">Orange Money</p>
   335→                      <p className="text-xs text-muted-foreground">Numéro pour recevoir les paiements</p>
   336→                    </div>
   337→                  </div>
   338→                  <div className="grid gap-3 sm:grid-cols-2">
   339→                    <div className="space-y-1.5">
   340→                      <Label htmlFor="om-phone" className="text-xs">Numéro Orange Money</Label>
   341→                      <Input
   342→                        id="om-phone"
   343→                        type="tel"
   344→                        placeholder="+237 6XX XXX XXX"
   345→                        value={omPhone}
   346→                        onChange={(e) => setOmPhone(e.target.value)}
   347→                        disabled={pmSaving}
   348→                        className="min-h-[44px]"
   349→                      />
   350→                    </div>
   351→                    <div className="space-y-1.5">
   352→                      <Label htmlFor="om-name" className="text-xs">Nom du titulaire</Label>
   353→                      <Input
   354→                        id="om-name"
   355→                        type="text"
   356→                        placeholder="Jean Kamga"
   357→                        value={omName}
   358→                        onChange={(e) => setOmName(e.target.value)}
   359→                        disabled={pmSaving}
   360→                        className="min-h-[44px]"
   361→                      />
   362→                    </div>
   363→                  </div>
   364→                </div>
   365→
   366→                <Separator />
   367→
   368→                {/* MTN MoMo */}
   369→                <div className="space-y-3">
   370→                  <div className="flex items-center gap-2">
   371→                    <div className="flex size-8 items-center justify-center rounded-lg bg-yellow-100 dark:bg-yellow-900/40">

   377→                    </div>
   378→                  </div>
   379→                  <div className="grid gap-3 sm:grid-cols-2">

   383→                        id="mtn-phone"
   384→                        type="tel"
   385→                        placeholder="+237 6XX XXX XXX"
   386→                        value={mtnPhone}
   387→                        onChange={(e) => setMtnPhone(e.target.value)}
   388→                        disabled={pmSaving}
   389→                        className="min-h-[44px]"
   390→                      />
   391→                    </div>
   392→                    <div className="space-y-1.5">
   393→                      <Label htmlFor="mtn-name" className="text-xs">Nom du titulaire</Label>
   394→                      <Input
   395→                        id="mtn-name"
   396→                        type="text"
   397→                        placeholder="Jean Kamga"
   398→                        value={mtnName}
   399→                        onChange={(e) => setMtnName(e.target.value)}
   400→                        disabled={pmSaving}
   401→                        className="min-h-[44px]"
   402→                      />
   403→                    </div>
   404→                  </div>
   405→                </div>
   406→
   407→                <Separator />
   408→
   409→                {/* Custom instructions */}

   417→                    value={pmInstructions}
   418→                    onChange={(e) => setPmInstructions(e.target.value)}
   419→                    disabled={pmSaving}
   420→                    rows={3}

   426→                  onClick={handleSavePaymentMethods}
   427→                  disabled={pmSaving}
   428→                  className="w-full min-h-[48px] bg-emerald-600 hover:bg-emerald-700 text-white sm:w-auto"
   429→                  size="lg"
   430→                >
   431→                  {pmSaving && <Loader2 size={16} className="mr-2 animate-spin" />}
   432→                  <CreditCard size={16} className="mr-2" />
   433→                  Enregistrer les paiements
   434→                </Button>
   435→              </>

   437→          </CardContent>
   438→        </Card>
   439→      </motion.div>
   440→
   441→      {/* Account section */}
   442→      <motion.div variants={itemVariants}>
   443→        <Card>
   444→          <CardHeader>
   445→            <CardTitle className="text-base flex items-center gap-2">
   446→              <Mail size={16} className="text-emerald-600" />
   447→              Compte
   448→            </CardTitle>
   449→            <CardDescription>
   450→              Informations liées à votre compte
   451→            </CardDescription>
   452→          </CardHeader>
   453→          <CardContent className="space-y-4">
   454→            <div className="flex items-center justify-between">

   457→                <p className="text-sm text-muted-foreground">
   458→                  {loadingEmail ? (
   459→                    <span className="inline-block w-48 h-4 bg-muted rounded animate-pulse" />

   474→                <p className="text-sm font-medium">Mot de passe</p>
   475→                <p className="text-xs text-muted-foreground">
   476→                  Modifiez votre mot de passe de connexion
   477→                </p>

   482→                  setNewPassword('');
   483→                  setConfirmPassword('');
   484→                  setPwdError('');
   485→                  setPwdDialogOpen(true);
   486→                }}
   487→              >
   488→                <Lock size={14} className="mr-2" />

   514→          <CardHeader>
