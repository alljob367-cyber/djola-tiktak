   381→-- ---------------------------------------------------------------
   382→-- 9a. sync_profile_subscription() — Trigger AFTER INSERT/UPDATE
   383→--      Synchronise les colonnes denormees du profil
   384→-- ---------------------------------------------------------------
   385→CREATE OR REPLACE FUNCTION public.sync_profile_subscription()
   386→RETURNS TRIGGER AS $$
   387→BEGIN
   388→  -- Cas 1 : l'abonnement est actif ou en essai (non expire)
   389→  IF NEW.status IN ('active', 'trialing') AND NEW.current_period_end >= now() THEN
   390→    UPDATE public.profiles SET
   391→      plan = NEW.plan_id,
   392→      subscription_status = NEW.status,
   393→      subscription_start = NEW.current_period_start,
   394→      subscription_end = NEW.current_period_end,
   395→      subscription_id = NEW.id,
   396→      updated_at = now()
   397→    WHERE id = NEW.profile_id;
   398→
   399→  -- Cas 2 : l'abonnement n'est plus actif
   400→  ELSIF NEW.status NOT IN ('active', 'trialing') THEN
   401→    UPDATE public.profiles SET
   402→      plan = 'starter',
   403→      subscription_status = CASE
   404→        WHEN NEW.status = 'upgraded' THEN (SELECT status FROM public.subscriptions WHERE profile_id = NEW.profile_id AND status IN ('active','trialing') LIMIT 1)
   405→        WHEN NEW.status = 'downgraded' THEN (SELECT status FROM public.subscriptions WHERE profile_id = NEW.profile_id AND status IN ('active','trialing') LIMIT 1)
   406→        ELSE NEW.status
   407→      END,
   408→      subscription_start = NULL,
   409→      subscription_end = NULL,
   410→      subscription_id = CASE
   411→        WHEN NEW.status IN ('upgraded', 'downgraded') THEN
   412→          (SELECT id FROM public.subscriptions WHERE profile_id = NEW.profile_id AND status IN ('active','trialing') LIMIT 1)
   413→        ELSE NULL
   414→      END,
   415→      updated_at = now()
   416→    WHERE id = NEW.profile_id;
   417→  END IF;
   418→
   419→  RETURN NEW;
   420→END;
   421→$$ LANGUAGE plpgsql SECURITY DEFINER;

   423→-- ---------------------------------------------------------------
   424→-- 9b. activate_subscription() — Activation d'un abonnement paye
   425→--      Utilise pg_advisory_lock pour la securite concurrentielle
   426→-- ---------------------------------------------------------------
   427→CREATE OR REPLACE FUNCTION public.activate_subscription(
   428→  p_profile_id UUID,
   429→  p_plan_id TEXT,
   430→  p_payment_id UUID,
   431→  p_duration_days INTEGER,
   432→  p_billing_period TEXT
   433→) RETURNS UUID AS $$
   434→DECLARE
   435→  v_new_sub_id UUID;
   436→  v_previous_status TEXT;
   437→  v_plan_name TEXT;
   438→BEGIN
   439→  -- Verrouillage advisory par profil (empeche les activations concurrentes)
   440→  PERFORM pg_advisory_lock(hashtext(p_profile_id::text));
   441→
   442→  BEGIN
   443→    -- Recuperer le nom du plan
   444→    SELECT name INTO v_plan_name FROM public.plans WHERE id = p_plan_id;
   445→
   446→    IF v_plan_name IS NULL THEN
   447→      RAISE EXCEPTION 'Plan introuvable : %', p_plan_id;
   448→    END IF;
   449→
   450→    -- Verifier et cloturer l'abonnement actuel/trialing existant
   451→    UPDATE public.subscriptions
   452→    SET
   453→      status = CASE
   454→        WHEN (SELECT tier_priority FROM public.plans WHERE id = plan_id) > (SELECT tier_priority FROM public.plans WHERE id = p_plan_id)
   455→          THEN 'downgraded'
   456→        ELSE 'upgraded'
   457→      END,
   458→      updated_at = now()
   459→    WHERE profile_id = p_profile_id
   460→      AND status IN ('active', 'trialing')
   461→    RETURNING status INTO v_previous_status;
   462→
   463→    -- Creer le nouvel abonnement
   464→    INSERT INTO public.subscriptions (
   465→      profile_id,
   466→      plan_id,
   467→      plan_name,
   468→      status,
   469→      billing_period,
   470→      current_period_start,
   471→      current_period_end,
   472→      payment_id
   473→    ) VALUES (
   474→      p_profile_id,
   475→      p_plan_id,
   476→      v_plan_name,
   477→      'active',
   478→      COALESCE(p_billing_period, 'monthly'),
   479→      now(),
   480→      now() + (COALESCE(p_duration_days, 30) || ' days')::interval,
   481→      p_payment_id
   482→    )
   483→    RETURNING id INTO v_new_sub_id;
   484→
   485→  EXCEPTION
   486→    WHEN OTHERS THEN
   487→      -- Toujours liberer le verrou en cas d'erreur
   488→      PERFORM pg_advisory_unlock(hashtext(p_profile_id::text));
   489→      RAISE;
   490→  END;
   491→
   492→  -- Liberer le verrou advisory
   493→  PERFORM pg_advisory_unlock(hashtext(p_profile_id::text));
   494→
   495→  RETURN v_new_sub_id;
   496→END;
   497→$$ LANGUAGE plpgsql SECURITY DEFINER;

   511→
   512→  BEGIN
   513→    -- Verifier qu'il n'y a pas deja un essai ou abonnement actif
   514→    IF EXISTS (
   515→      SELECT 1 FROM public.subscriptions
   516→      WHERE profile_id = p_profile_id AND status IN ('active', 'trialing')
   517→    ) THEN
   518→      PERFORM pg_advisory_unlock(hashtext(p_profile_id::text));
   519→      RAISE EXCEPTION 'Un abonnement ou essai est deja actif pour ce profil';
   520→    END IF;
   521→
   522→    -- Verifier qu'il n'y a pas eu un essai precedemment
   523→    IF EXISTS (
   524→      SELECT 1 FROM public.subscriptions
   525→      WHERE profile_id = p_profile_id AND status IN ('expired', 'cancelled', 'upgraded', 'downgraded')
   526→        AND trial_end IS NOT NULL
   527→    ) THEN
   528→      PERFORM pg_advisory_unlock(hashtext(p_profile_id::text));
   529→      RAISE EXCEPTION 'Un essai a deja ete utilise pour ce profil';
   530→    END IF;
   531→
   532→    -- Recuperer le nom du plan starter
   533→    SELECT name INTO v_plan_name FROM public.plans WHERE id = 'starter';
   534→
   535→    -- Creer l'abonnement d'essai
   536→    INSERT INTO public.subscriptions (
   537→      profile_id,
   538→      plan_id,
   539→      plan_name,
   540→      status,
   541→      billing_period,
   542→      current_period_start,
   543→      current_period_end,
   544→      trial_start,
   545→      trial_end
   546→    ) VALUES (
   547→      p_profile_id,
   548→      'starter',
   549→      COALESCE(v_plan_name, 'Starter'),
   550→      'trialing',
   551→      'monthly',
   552→      now(),
   553→      now() + INTERVAL '7 days',
   554→      now(),
   555→      now() + INTERVAL '7 days'
   556→    )
   557→    RETURNING id INTO v_new_sub_id;
   558→
   559→  EXCEPTION
   560→    WHEN OTHERS THEN
   561→      PERFORM pg_advisory_unlock(hashtext(p_profile_id::text));
   562→      RAISE;
   563→  END;
   564→
   565→  -- Liberer le verrou advisory
   566→  PERFORM pg_advisory_unlock(hashtext(p_profile_id::text));
   567→
   568→  RETURN v_new_sub_id;
   569→END;
   570→$$ LANGUAGE plpgsql SECURITY DEFINER;
   571→
   572→-- ---------------------------------------------------------------
   573→-- 9d. consume_voice_credit() — Fonction CRITIQUE de consommation
   574→--      100% atomique et securisee contre la concurrence
   575→-- ---------------------------------------------------------------
   576→CREATE OR REPLACE FUNCTION public.consume_voice_credit(
   577→  p_profile_id UUID,
   578→  p_reference_id TEXT,
   579→  p_metadata JSONB
   580→) RETURNS TABLE (
   581→  allowed BOOLEAN,
   582→  usage_record_id UUID,
   583→  remaining INTEGER,
   584→  message TEXT
   585→) AS $$
   586→DECLARE
   587→  v_subscription_id UUID;
   588→  v_limit_value INTEGER;
   589→  v_current_usage BIGINT;
   590→  v_cost_per_unit NUMERIC(10,6);
   591→  v_new_record_id UUID;
   592→  v_remaining_int INTEGER;
   593→BEGIN
   594→  -- Verrouillage advisory par profil (critique pour la concurrence)
   595→  PERFORM pg_advisory_lock(hashtext(p_profile_id::text));
   596→
   597→  BEGIN
   598→    -- Trouver l'abonnement actif ou en essai
   599→    SELECT id INTO v_subscription_id
   600→    FROM public.subscriptions
   601→    WHERE profile_id = p_profile_id
   602→      AND status IN ('active', 'trialing')
   603→      AND current_period_end >= now()
   604→    LIMIT 1;
   605→
   606→    -- Pas d'abonnement actif
   607→    IF v_subscription_id IS NULL THEN
   608→      PERFORM pg_advisory_unlock(hashtext(p_profile_id::text));
   609→      allowed := false;
   610→      usage_record_id := NULL;
   611→      remaining := 0;
   612→      message := 'Aucun abonnement actif';
   613→      RETURN NEXT;
   614→      RETURN;
   615→    END IF;
   616→
   617→    -- Recuperer la limite de credits vocaux pour le plan
   618→    SELECT pl.limit_value, pl.cost_per_unit INTO v_limit_value, v_cost_per_unit
   619→    FROM public.plan_limits pl
   620→    JOIN public.subscriptions s ON s.plan_id = pl.plan_id
   621→    WHERE s.id = v_subscription_id
   622→      AND pl.limit_key = 'voice_credits'
   623→    LIMIT 1;
   624→
   625→    -- Limite non trouvee (ne devrait pas arriver)
   626→    IF v_limit_value IS NULL THEN
   627→      PERFORM pg_advisory_unlock(hashtext(p_profile_id::text));
   628→      allowed := false;
   629→      usage_record_id := NULL;
   630→      remaining := 0;
   631→      message := 'Limite de credits non configuree';
   632→      RETURN NEXT;
   633→      RETURN;
   634→    END IF;
   635→
   636→    -- Limite illimitee (-1) : toujours autorise
   637→    IF v_limit_value = -1 THEN
   638→      INSERT INTO public.usage_records (
   639→        profile_id, subscription_id, usage_type, quantity, provider,
   640→        reference_id, status, estimated_cost, metadata
   641→      ) VALUES (
   642→        p_profile_id, v_subscription_id, 'voice_generation', 1, 'elevenlabs',
   643→        p_reference_id, 'reserved', v_cost_per_unit, COALESCE(p_metadata, '{}'::jsonb)
   644→      )
   645→      RETURNING id INTO v_new_record_id;
   646→
   647→      PERFORM pg_advisory_unlock(hashtext(p_profile_id::text));
   648→      allowed := true;
   649→      usage_record_id := v_new_record_id;
   650→      remaining := -1;
   651→      message := 'ok';
   652→      RETURN NEXT;
   653→      RETURN;
   654→    END IF;
   655→
   656→    -- Compter la consommation courante (reserved + completed) pour la periode
   657→    SELECT COUNT(*) INTO v_current_usage
   658→    FROM public.usage_records
   659→    WHERE profile_id = p_profile_id
   660→      AND usage_type = 'voice_generation'
   661→      AND status IN ('reserved', 'completed')
   662→      AND created_at >= (
   663→        SELECT current_period_start FROM public.subscriptions WHERE id = v_subscription_id
   664→      );
   665→
   666→    -- Verifier si le quota est epuise
   667→    IF v_current_usage >= v_limit_value THEN
   668→      PERFORM pg_advisory_unlock(hashtext(p_profile_id::text));
   669→      allowed := false;
   670→      usage_record_id := NULL;
   671→      remaining := 0;
   672→      message := 'quota epuise';
   673→      RETURN NEXT;
   674→      RETURN;
   675→    END IF;
   676→
   677→    -- Creer l'enregistrement de consommation
   678→    INSERT INTO public.usage_records (
   679→      profile_id, subscription_id, usage_type, quantity, provider,
   680→      reference_id, status, estimated_cost, metadata
   681→    ) VALUES (
   682→      p_profile_id, v_subscription_id, 'voice_generation', 1, 'elevenlabs',
   683→      p_reference_id, 'reserved', v_cost_per_unit, COALESCE(p_metadata, '{}'::jsonb)
   684→    )
   685→    RETURNING id INTO v_new_record_id;
   686→
   687→    v_remaining_int := (v_limit_value - v_current_usage - 1)::INTEGER;
   688→
   689→    PERFORM pg_advisory_unlock(hashtext(p_profile_id::text));
   690→    allowed := true;
   691→    usage_record_id := v_new_record_id;
   692→    remaining := v_remaining_int;
   693→    message := 'ok';
   694→    RETURN NEXT;
   695→    RETURN;
   696→
   697→  EXCEPTION
   698→    WHEN OTHERS THEN
   699→      PERFORM pg_advisory_unlock(hashtext(p_profile_id::text));
   700→      RAISE;
   701→  END;
   702→END;
   703→$$ LANGUAGE plpgsql SECURITY DEFINER;
   704→
   705→-- ---------------------------------------------------------------
   706→-- 9e. complete_voice_credit() — Confirmation ou remboursement
   707→--      Mecanisme de compensation pour les credits reserves
   708→-- ---------------------------------------------------------------
   709→CREATE OR REPLACE FUNCTION public.complete_voice_credit(
   710→  p_usage_record_id UUID,
   711→  p_success BOOLEAN
   712→) RETURNS VOID AS $$
   713→BEGIN
   714→  IF p_success THEN
   715→    UPDATE public.usage_records
   716→    SET status = 'completed'
   717→    WHERE id = p_usage_record_id
   718→      AND status = 'reserved';
   719→  ELSE
   720→    -- En cas d'echec, le credit est rembourse (status = refunded)
   721→    UPDATE public.usage_records
   722→    SET status = 'refunded'
   723→    WHERE id = p_usage_record_id
   724→      AND status = 'reserved';
   725→  END IF;
   726→END;
   727→$$ LANGUAGE plpgsql SECURITY DEFINER;

   729→-- ---------------------------------------------------------------
   730→-- 9f. expire_subscriptions() — Expiration batch des abonnements
   731→--      A appeler via un cron (pg_cron ou Edge Function)
   732→-- ---------------------------------------------------------------
   733→CREATE OR REPLACE FUNCTION public.expire_subscriptions()
   734→RETURNS INTEGER AS $$
   735→DECLARE
   736→  v_count INTEGER := 0;
   737→BEGIN
   738→  -- Verrouillage global pour eviter les executions concurrentes du batch
   739→  PERFORM pg_advisory_lock(54321);
   740→
   741→  -- Expirer les abonnements actifs/trialing dont la periode est terminee
   742→  UPDATE public.subscriptions
   743→  SET
   744→    status = 'expired',
   745→    updated_at = now()
   746→  WHERE status IN ('active', 'trialing')
   747→    AND current_period_end < now();
   748→
   749→  GET DIAGNOSTICS v_count = ROW_COUNT;
   750→
   751→  -- Expirer specifiquement les essais termines
   752→  UPDATE public.subscriptions
   753→  SET
   754→    status = 'expired',
   755→    updated_at = now()
   756→  WHERE status = 'trialing'
   757→    AND trial_end IS NOT NULL
   758→    AND trial_end < now();
   759→
   760→  -- Le trigger trg_sync_profile_subscription va automatiquement
   761→  -- synchroniser les profils concernes
   762→
   763→  -- Liberer le verrou
   764→  PERFORM pg_advisory_unlock(54321);
   765→
   766→  RETURN v_count;
   767→END;
   768→$$ LANGUAGE plpgsql SECURITY DEFINER;

   770→-- ---------------------------------------------------------------
   771→-- 9g. get_usage_summary() — Resume de consommation par profil
   772→--      Retourne une ligne par type de limite avec usage courant
   773→-- ---------------------------------------------------------------
   774→CREATE OR REPLACE FUNCTION public.get_usage_summary(
   775→  p_profile_id UUID
   776→) RETURNS TABLE (
   777→  limit_key TEXT,
   778→  limit_value INTEGER,
   779→  limit_label TEXT,
   780→  current_usage BIGINT,
   781→  remaining INTEGER,
   782→  is_unlimited BOOLEAN
   783→) AS $$
   784→BEGIN
   785→  RETURN QUERY
   786→  SELECT
   787→    pl.limit_key,
   788→    pl.limit_value,
   789→    pl.unit_label,
   790→    COALESCE(ur.usage_count, 0) AS current_usage,
   791→    CASE
   792→      WHEN pl.limit_value = -1 THEN -1
   793→      WHEN pl.limit_value IS NULL THEN 0
   794→      ELSE GREATEST((pl.limit_value - COALESCE(ur.usage_count, 0))::INTEGER, 0)
   795→    END AS remaining,
   796→    (pl.limit_value = -1) AS is_unlimited
   797→  FROM public.plan_limits pl
   798→  LEFT JOIN LATERAL (
   799→    SELECT COUNT(*) AS usage_count
   800→    FROM public.usage_records ur2
   801→    WHERE ur2.profile_id = p_profile_id
   802→      AND ur2.usage_type = pl.limit_key
   803→      AND ur2.status IN ('reserved', 'completed')
   804→      AND ur2.created_at >= (
   805→        SELECT COALESCE(
   806→          MIN(s.current_period_start),
   807→          now() - INTERVAL '30 days'
   808→        )
   809→        FROM public.subscriptions s
   810→        WHERE s.profile_id = p_profile_id
   811→          AND s.status IN ('active', 'trialing')
   812→      )
   813→  ) ur ON true
   814→  WHERE pl.plan_id = (
   815→    SELECT COALESCE(
   816→      (SELECT plan_id FROM public.subscriptions
   817→       WHERE profile_id = p_profile_id AND status IN ('active', 'trialing')
   818→       LIMIT 1),
   819→      (SELECT plan FROM public.profiles WHERE id = p_profile_id),
   820→      'starter'
   821→    )
   822→  );
   823→END;
   824→$$ LANGUAGE plpgsql SECURITY DEFINER STABLE;

  1010→-- 13a. v_subscription_summary : resume de l'abonnement par profil
  1011→-- ---------------------------------------------------------------
  1012→CREATE OR REPLACE VIEW public.v_subscription_summary AS
  1013→SELECT
  1014→  p.id AS profile_id,
  1015→  p.business_name,
  1016→  p.plan AS current_plan,
  1017→  p.subscription_status,
  1018→  p.subscription_start,
  1019→  p.subscription_end,
  1020→  p.subscription_id,
  1021→  CASE
  1022→    WHEN p.subscription_end IS NULL THEN NULL
  1023→    WHEN p.subscription_end < now() THEN 0
  1024→    ELSE GREATEST(EXTRACT(DAY FROM p.subscription_end - now())::INTEGER, 0)
  1025→  END AS days_remaining,
  1026→  CASE
  1027→    WHEN p.subscription_status IN ('active', 'trialing') AND p.subscription_end >= now() THEN true
  1028→    ELSE false
  1029→  END AS is_active,
  1030→  s.trial_start,
  1031→  s.trial_end,
  1032→  CASE
  1033→    WHEN s.trial_end IS NOT NULL AND s.trial_end > now() THEN true
  1034→    ELSE false
  1035→  END AS is_trial,
  1036→  CASE
  1037→    WHEN s.trial_end IS NOT NULL AND s.trial_end > now()
  1038→      THEN GREATEST(EXTRACT(DAY FROM s.trial_end - now())::INTEGER, 0)
  1039→    ELSE NULL
  1040→  END AS trial_days_remaining,
  1041→  pl.name AS plan_display_name,
  1042→  pl.price_monthly,
  1043→  pl.price_yearly,
  1044→  s.billing_period,
  1045→  s.cancel_at_period_end
  1046→FROM public.profiles p
  1047→LEFT JOIN public.subscriptions s ON s.id = p.subscription_id
  1048→LEFT JOIN public.plans pl ON pl.id = p.plan;

  1050→-- ---------------------------------------------------------------
  1051→-- 13b. v_monthly_revenue : revenus mensuels par plan (admin)
  1052→-- ---------------------------------------------------------------
  1053→CREATE OR REPLACE VIEW public.v_monthly_revenue AS
  1054→SELECT
  1055→  TO_CHAR(pay.created_at, 'YYYY-MM') AS month,
  1056→  pay.plan_id,
  1057→  pay.plan_name,
  1058→  pay.billing_period,
  1059→  COUNT(*) AS payment_count,
  1060→  SUM(pay.amount) AS total_amount,
  1061→  pay.currency,
  1062→  COUNT(*) FILTER (WHERE pay.status = 'completed') AS completed_count,
  1063→  SUM(pay.amount) FILTER (WHERE pay.status = 'completed') AS completed_amount,
  1064→  COUNT(*) FILTER (WHERE pay.status = 'pending') AS pending_count,
  1065→  COUNT(*) FILTER (WHERE pay.status = 'failed') AS failed_count
  1066→FROM public.payments pay
  1067→GROUP BY
  1068→  TO_CHAR(pay.created_at, 'YYYY-MM'),
  1069→  pay.plan_id,
  1070→  pay.plan_name,
  1071→  pay.billing_period,
  1072→  pay.currency
  1073→ORDER BY month DESC, pay.plan_id;

  1076→-- 13c. v_expiring_soon : abonnements expirant dans les 7 prochains jours
  1077→-- ---------------------------------------------------------------
  1078→CREATE OR REPLACE VIEW public.v_expiring_soon AS
  1079→SELECT
  1080→  s.id AS subscription_id,
  1081→  s.profile_id,
  1082→  p.business_name,
  1083→  p.email,
  1084→  p.phone,
  1085→  s.plan_id,
  1086→  s.plan_name,
  1087→  s.status,
  1088→  s.billing_period,
  1089→  s.current_period_start,
  1090→  s.current_period_end,
  1091→  GREATEST(EXTRACT(DAY FROM s.current_period_end - now())::INTEGER, 0) AS days_remaining,
  1092→  s.cancel_at_period_end
  1093→FROM public.subscriptions s
  1094→JOIN public.profiles p ON p.id = s.profile_id
  1095→WHERE s.status IN ('active', 'trialing')
  1096→  AND s.current_period_end > now()
  1097→  AND s.current_period_end <= now() + INTERVAL '7 days'
  1098→ORDER BY s.current_period_end ASC;

  1100→-- ---------------------------------------------------------------
  1101→-- 13d. v_usage_dashboard : tableau de bord de consommation par profil
  1102→-- ---------------------------------------------------------------
  1103→CREATE OR REPLACE VIEW public.v_usage_dashboard AS
  1104→SELECT
  1105→  ur.profile_id,
  1106→  p.business_name,
  1107→  ur.usage_type,
  1108→  ur.status,
  1109→  COUNT(*) AS record_count,
  1110→  SUM(ur.quantity) AS total_quantity,
  1111→  SUM(ur.estimated_cost) AS total_estimated_cost,
  1112→ MIN(ur.created_at) AS first_use,
  1113→  MAX(ur.created_at) AS last_use
  1114→FROM public.usage_records ur
  1115→JOIN public.profiles p ON p.id = ur.profile_id
  1116→WHERE ur.created_at >= (
  1117→  SELECT COALESCE(MIN(s.current_period_start), now() - INTERVAL '30 days')
  1118→  FROM public.subscriptions s
  1119→  WHERE s.profile_id = ur.profile_id AND s.status IN ('active', 'trialing')
  1120→)
  1121→GROUP BY ur.profile_id, p.business_name, ur.usage_type, ur.status
  1122→ORDER BY ur.profile_id, ur.usage_type, ur.status;
