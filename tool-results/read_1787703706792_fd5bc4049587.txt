   381→   381→-- ---------------------------------------------------------------
   382→   382→-- 9a. sync_profile_subscription() — Trigger AFTER INSERT/UPDATE
   383→   383→--      Synchronise les colonnes denormees du profil
   384→   384→-- ---------------------------------------------------------------
   385→   385→CREATE OR REPLACE FUNCTION public.sync_profile_subscription()
   386→   386→RETURNS TRIGGER AS $$
   387→   387→BEGIN
   388→   388→  -- Cas 1 : l'abonnement est actif ou en essai (non expire)
   389→   389→  IF NEW.status IN ('active', 'trialing') AND NEW.current_period_end >= now() THEN
   390→   390→    UPDATE public.profiles SET
   391→   391→      plan = NEW.plan_id,
   392→   392→      subscription_status = NEW.status,
   393→   393→      subscription_start = NEW.current_period_start,
   394→   394→      subscription_end = NEW.current_period_end,
   395→   395→      subscription_id = NEW.id,
   396→   396→      updated_at = now()
   397→   397→    WHERE id = NEW.profile_id;
   398→   398→
   399→   399→  -- Cas 2 : l'abonnement n'est plus actif
   400→   400→  ELSIF NEW.status NOT IN ('active', 'trialing') THEN
   401→   401→    UPDATE public.profiles SET
   402→   402→      plan = 'starter',
   403→   403→      subscription_status = CASE
   404→   404→        WHEN NEW.status = 'upgraded' THEN (SELECT status FROM public.subscriptions WHERE profile_id = NEW.profile_id AND status IN ('active','trialing') LIMIT 1)
   405→   405→        WHEN NEW.status = 'downgraded' THEN (SELECT status FROM public.subscriptions WHERE profile_id = NEW.profile_id AND status IN ('active','trialing') LIMIT 1)
   406→   406→        ELSE NEW.status
   407→   407→      END,
   408→   408→      subscription_start = NULL,
   409→   409→      subscription_end = NULL,
   410→   410→      subscription_id = CASE
   411→   411→        WHEN NEW.status IN ('upgraded', 'downgraded') THEN
   412→   412→          (SELECT id FROM public.subscriptions WHERE profile_id = NEW.profile_id AND status IN ('active','trialing') LIMIT 1)
   413→   413→        ELSE NULL
   414→   414→      END,
   415→   415→      updated_at = now()
   416→   416→    WHERE id = NEW.profile_id;
   417→   417→  END IF;
   418→   418→
   419→   419→  RETURN NEW;
   420→   420→END;
   421→   421→$$ LANGUAGE plpgsql SECURITY DEFINER;

   423→   423→-- ---------------------------------------------------------------
   424→   424→-- 9b. activate_subscription() — Activation d'un abonnement paye
   425→   425→--      Utilise pg_advisory_lock pour la securite concurrentielle
   426→   426→-- ---------------------------------------------------------------
   427→   427→CREATE OR REPLACE FUNCTION public.activate_subscription(
   428→   428→  p_profile_id UUID,
   429→   429→  p_plan_id TEXT,
   430→   430→  p_payment_id UUID,
   431→   431→  p_duration_days INTEGER,
   432→   432→  p_billing_period TEXT
   433→   433→) RETURNS UUID AS $$
   434→   434→DECLARE
   435→   435→  v_new_sub_id UUID;
   436→   436→  v_previous_status TEXT;
   437→   437→  v_plan_name TEXT;
   438→   438→BEGIN
   439→   439→  -- Verrouillage advisory par profil (empeche les activations concurrentes)
   440→   440→  PERFORM pg_advisory_lock(hashtext(p_profile_id::text));
   441→   441→
   442→   442→  BEGIN
   443→   443→    -- Recuperer le nom du plan
   444→   444→    SELECT name INTO v_plan_name FROM public.plans WHERE id = p_plan_id;
   445→   445→
   446→   446→    IF v_plan_name IS NULL THEN
   447→   447→      RAISE EXCEPTION 'Plan introuvable : %', p_plan_id;
   448→   448→    END IF;
   449→   449→
   450→   450→    -- Verifier et cloturer l'abonnement actuel/trialing existant
   451→   451→    UPDATE public.subscriptions
   452→   452→    SET
   453→   453→      status = CASE
   454→   454→        WHEN (SELECT tier_priority FROM public.plans WHERE id = plan_id) > (SELECT tier_priority FROM public.plans WHERE id = p_plan_id)
   455→   455→          THEN 'downgraded'
   456→   456→        ELSE 'upgraded'
   457→   457→      END,
   458→   458→      updated_at = now()
   459→   459→    WHERE profile_id = p_profile_id
   460→   460→      AND status IN ('active', 'trialing')
   461→   461→    RETURNING status INTO v_previous_status;
   462→   462→
   463→   463→    -- Creer le nouvel abonnement
   464→   464→    INSERT INTO public.subscriptions (
   465→   465→      profile_id,
   466→   466→      plan_id,
   467→   467→      plan_name,
   468→   468→      status,
   469→   469→      billing_period,
   470→   470→      current_period_start,
   471→   471→      current_period_end,
   472→   472→      payment_id
   473→   473→    ) VALUES (
   474→   474→      p_profile_id,
   475→   475→      p_plan_id,
   476→   476→      v_plan_name,
   477→   477→      'active',
   478→   478→      COALESCE(p_billing_period, 'monthly'),
   479→   479→      now(),
   480→   480→      now() + (COALESCE(p_duration_days, 30) || ' days')::interval,
   481→   481→      p_payment_id
   482→   482→    )
   483→   483→    RETURNING id INTO v_new_sub_id;
   484→   484→
   485→   485→  EXCEPTION
   486→   486→    WHEN OTHERS THEN
   487→   487→      -- Toujours liberer le verrou en cas d'erreur
   488→   488→      PERFORM pg_advisory_unlock(hashtext(p_profile_id::text));
   489→   489→      RAISE;
   490→   490→  END;
   491→   491→
   492→   492→  -- Liberer le verrou advisory
   493→   493→  PERFORM pg_advisory_unlock(hashtext(p_profile_id::text));
   494→   494→
   495→   495→  RETURN v_new_sub_id;
   496→   496→END;
   497→   497→$$ LANGUAGE plpgsql SECURITY DEFINER;

   499→   499→-- ---------------------------------------------------------------
   500→   500→-- 9c. start_trial() — Demarrage d'un essai de 7 jours
   501→   501→-- ---------------------------------------------------------------
   502→   502→CREATE OR REPLACE FUNCTION public.start_trial(
   503→   503→  p_profile_id UUID
   504→   504→) RETURNS UUID AS $$
   505→   505→DECLARE
   506→   506→  v_new_sub_id UUID;
   507→   507→  v_plan_name TEXT;
   508→   508→BEGIN
   509→   509→  -- Verrouillage advisory par profil
   510→   510→  PERFORM pg_advisory_lock(hashtext(p_profile_id::text));
   511→   511→
   512→   512→  BEGIN
   513→   513→    -- Verifier qu'il n'y a pas deja un essai ou abonnement actif
   514→   514→    IF EXISTS (
   515→   515→      SELECT 1 FROM public.subscriptions
   516→   516→      WHERE profile_id = p_profile_id AND status IN ('active', 'trialing')
   517→   517→    ) THEN
   518→   518→      PERFORM pg_advisory_unlock(hashtext(p_profile_id::text));

   520→   520→    END IF;

   523→   523→    IF EXISTS (
   524→   524→      SELECT 1 FROM public.subscriptions
   525→   525→      WHERE profile_id = p_profile_id AND status IN ('expired', 'cancelled', 'upgraded', 'downgraded')
   526→   526→        AND trial_end IS NOT NULL
   527→   527→    ) THEN
   528→   528→      PERFORM pg_advisory_unlock(hashtext(p_profile_id::text));
   529→   529→      RAISE EXCEPTION 'Un essai a deja ete utilise pour ce profil';
   530→   530→    END IF;
   531→   531→
   532→   532→    -- Recuperer le nom du plan starter
   533→   533→    SELECT name INTO v_plan_name FROM public.plans WHERE id = 'starter';
   534→   534→
   535→   535→    -- Creer l'abonnement d'essai
   536→   536→    INSERT INTO public.subscriptions (
   537→   537→      profile_id,
   538→   538→      plan_id,
   539→   539→      plan_name,
   540→   540→      status,
   541→   541→      billing_period,
   542→   542→      current_period_start,
   543→   543→      current_period_end,
   544→   544→      trial_start,
   545→   545→      trial_end
   546→   546→    ) VALUES (
   547→   547→      p_profile_id,
   548→   548→      'starter',
   549→   549→      COALESCE(v_plan_name, 'Starter'),
   550→   550→      'trialing',
   551→   551→      'monthly',
   552→   552→      now(),
   553→   553→      now() + INTERVAL '7 days',
   554→   554→      now(),
   555→   555→      now() + INTERVAL '7 days'
   556→   556→    )
   557→   557→    RETURNING id INTO v_new_sub_id;
   558→   558→
   559→   559→  EXCEPTION
   560→   560→    WHEN OTHERS THEN
   561→   561→      PERFORM pg_advisory_unlock(hashtext(p_profile_id::text));
   562→   562→      RAISE;
   563→   563→  END;
   564→   564→
   565→   565→  -- Liberer le verrou advisory
   566→   566→  PERFORM pg_advisory_unlock(hashtext(p_profile_id::text));
   567→   567→
   568→   568→  RETURN v_new_sub_id;
   569→   569→END;
   570→   570→$$ LANGUAGE plpgsql SECURITY DEFINER;

   572→   572→-- ---------------------------------------------------------------
   573→   573→-- 9d. consume_voice_credit() — Fonction CRITIQUE de consommation
   574→   574→--      100% atomique et securisee contre la concurrence
   575→   575→-- ---------------------------------------------------------------
   576→   576→CREATE OR REPLACE FUNCTION public.consume_voice_credit(
   577→   577→  p_profile_id UUID,
   578→   578→  p_reference_id TEXT,
   579→   579→  p_metadata JSONB
   580→   580→) RETURNS TABLE (
   581→   581→  allowed BOOLEAN,
   582→   582→  usage_record_id UUID,
   583→   583→  remaining INTEGER,
   584→   584→  message TEXT
   585→   585→) AS $$
   586→   586→DECLARE
   587→   587→  v_subscription_id UUID;
   588→   588→  v_limit_value INTEGER;
   589→   589→  v_current_usage BIGINT;
   590→   590→  v_cost_per_unit NUMERIC(10,6);
   591→   591→  v_new_record_id UUID;
   592→   592→  v_remaining_int INTEGER;
   593→   593→BEGIN
   594→   594→  -- Verrouillage advisory par profil (critique pour la concurrence)
   595→   595→  PERFORM pg_advisory_lock(hashtext(p_profile_id::text));
   596→   596→
   597→   597→  BEGIN
   598→   598→    -- Trouver l'abonnement actif ou en essai
   599→   599→    SELECT id INTO v_subscription_id
   600→   600→    FROM public.subscriptions
   601→   601→    WHERE profile_id = p_profile_id
   602→   602→      AND status IN ('active', 'trialing')
   603→   603→      AND current_period_end >= now()
   604→   604→    LIMIT 1;
   605→   605→
   606→   606→    -- Pas d'abonnement actif
   607→   607→    IF v_subscription_id IS NULL THEN
   608→   608→      PERFORM pg_advisory_unlock(hashtext(p_profile_id::text));
   609→   609→      allowed := false;
   610→   610→      usage_record_id := NULL;
   611→   611→      remaining := 0;
   612→   612→      message := 'Aucun abonnement actif';
   613→   613→      RETURN NEXT;
   614→   614→      RETURN;
   615→   615→    END IF;
   616→   616→
   617→   617→    -- Recuperer la limite de credits vocaux pour le plan
   618→   618→    SELECT pl.limit_value, pl.cost_per_unit INTO v_limit_value, v_cost_per_unit
   619→   619→    FROM public.plan_limits pl
   620→   620→    JOIN public.subscriptions s ON s.plan_id = pl.plan_id
   621→   621→    WHERE s.id = v_subscription_id
   622→   622→      AND pl.limit_key = 'voice_credits'
   623→   623→    LIMIT 1;
   624→   624→
   625→   625→    -- Limite non trouvee (ne devrait pas arriver)
   626→   626→    IF v_limit_value IS NULL THEN
   627→   627→      PERFORM pg_advisory_unlock(hashtext(p_profile_id::text));
   628→   628→      allowed := false;
   629→   629→      usage_record_id := NULL;
   630→   630→      remaining := 0;
   631→   631→      message := 'Limite de credits non configuree';
   632→   632→      RETURN NEXT;
   633→   633→      RETURN;
   634→   634→    END IF;
   635→   635→
   636→   636→    -- Limite illimitee (-1) : toujours autorise
   637→   637→    IF v_limit_value = -1 THEN
   638→   638→      INSERT INTO public.usage_records (
   639→   639→        profile_id, subscription_id, usage_type, quantity, provider,
   640→   640→        reference_id, status, estimated_cost, metadata
   641→   641→      ) VALUES (
   642→   642→        p_profile_id, v_subscription_id, 'voice_generation', 1, 'elevenlabs',
   643→   643→        p_reference_id, 'reserved', v_cost_per_unit, COALESCE(p_metadata, '{}'::jsonb)
   644→   644→      )
   645→   645→      RETURNING id INTO v_new_record_id;
   646→   646→
   647→   647→      PERFORM pg_advisory_unlock(hashtext(p_profile_id::text));
   648→   648→      allowed := true;
   649→   649→      usage_record_id := v_new_record_id;
   650→   650→      remaining := -1;
   651→   651→      message := 'ok';
   652→   652→      RETURN NEXT;
   653→   653→      RETURN;
   654→   654→    END IF;
   655→   655→
   656→   656→    -- Compter la consommation courante (reserved + completed) pour la periode
   657→   657→    SELECT COUNT(*) INTO v_current_usage
   658→   658→    FROM public.usage_records
   659→   659→    WHERE profile_id = p_profile_id
   660→   660→      AND usage_type = 'voice_generation'
   661→   661→      AND status IN ('reserved', 'completed')
   662→   662→      AND created_at >= (
   663→   663→        SELECT current_period_start FROM public.subscriptions WHERE id = v_subscription_id
   664→   664→      );
   665→   665→
   666→   666→    -- Verifier si le quota est epuise
   667→   667→    IF v_current_usage >= v_limit_value THEN
   668→   668→      PERFORM pg_advisory_unlock(hashtext(p_profile_id::text));
   669→   669→      allowed := false;
   670→   670→      usage_record_id := NULL;
   671→   671→      remaining := 0;
   672→   672→      message := 'quota epuise';
   673→   673→      RETURN NEXT;
   674→   674→      RETURN;
   675→   675→    END IF;
   676→   676→
   677→   677→    -- Creer l'enregistrement de consommation
   678→   678→    INSERT INTO public.usage_records (
   679→   679→      profile_id, subscription_id, usage_type, quantity, provider,
   680→   680→      reference_id, status, estimated_cost, metadata
   681→   681→    ) VALUES (
   682→   682→      p_profile_id, v_subscription_id, 'voice_generation', 1, 'elevenlabs',
   683→   683→      p_reference_id, 'reserved', v_cost_per_unit, COALESCE(p_metadata, '{}'::jsonb)
   684→   684→    )
   685→   685→    RETURNING id INTO v_new_record_id;
   686→   686→
   687→   687→    v_remaining_int := (v_limit_value - v_current_usage - 1)::INTEGER;
   688→   688→
   689→   689→    PERFORM pg_advisory_unlock(hashtext(p_profile_id::text));
   690→   690→    allowed := true;
   691→   691→    usage_record_id := v_new_record_id;
   692→   692→    remaining := v_remaining_int;
   693→   693→    message := 'ok';
   694→   694→    RETURN NEXT;
   695→   695→    RETURN;
   696→   696→
   697→   697→  EXCEPTION
   698→   698→    WHEN OTHERS THEN
   699→   699→      PERFORM pg_advisory_unlock(hashtext(p_profile_id::text));
   700→   700→      RAISE;
   701→   701→  END;
   702→   702→END;
   703→   703→$$ LANGUAGE plpgsql SECURITY DEFINER;
   704→   704→
   705→   705→-- ---------------------------------------------------------------
   706→   706→-- 9e. complete_voice_credit() — Confirmation ou remboursement
   707→   707→--      Mecanisme de compensation pour les credits reserves
   708→   708→-- ---------------------------------------------------------------
   709→   709→CREATE OR REPLACE FUNCTION public.complete_voice_credit(
   710→   710→  p_usage_record_id UUID,
   711→   711→  p_success BOOLEAN
   712→   712→) RETURNS VOID AS $$
   713→   713→BEGIN
   714→   714→  IF p_success THEN
   715→   715→    UPDATE public.usage_records
   716→   716→    SET status = 'completed'
   717→   717→    WHERE id = p_usage_record_id
   718→   718→      AND status = 'reserved';
   719→   719→  ELSE
   720→   720→    -- En cas d'echec, le credit est rembourse (status = refunded)
   721→   721→    UPDATE public.usage_records
   722→   722→    SET status = 'refunded'
   723→   723→    WHERE id = p_usage_record_id
   724→   724→      AND status = 'reserved';
   725→   725→  END IF;
   726→   726→END;
   727→   727→$$ LANGUAGE plpgsql SECURITY DEFINER;

   729→   729→-- ---------------------------------------------------------------
   730→   730→-- 9f. expire_subscriptions() — Expiration batch des abonnements
   731→   731→--      A appeler via un cron (pg_cron ou Edge Function)
   732→   732→-- ---------------------------------------------------------------
   733→   733→CREATE OR REPLACE FUNCTION public.expire_subscriptions()
   734→   734→RETURNS INTEGER AS $$
   735→   735→DECLARE
   736→   736→  v_count INTEGER := 0;
   737→   737→BEGIN
   738→   738→  -- Verrouillage global pour eviter les executions concurrentes du batch
   739→   739→  PERFORM pg_advisory_lock(54321);
   740→   740→
   741→   741→  -- Expirer les abonnements actifs/trialing dont la periode est terminee
   742→   742→  UPDATE public.subscriptions
   743→   743→  SET
   744→   744→    status = 'expired',
   745→   745→    updated_at = now()
   746→   746→  WHERE status IN ('active', 'trialing')
   747→   747→    AND current_period_end < now();
   748→   748→
   749→   749→  GET DIAGNOSTICS v_count = ROW_COUNT;
   750→   750→
   751→   751→  -- Expirer specifiquement les essais termines
   752→   752→  UPDATE public.subscriptions
   753→   753→  SET
   754→   754→    status = 'expired',
   755→   755→    updated_at = now()
   756→   756→  WHERE status = 'trialing'
   757→   757→    AND trial_end IS NOT NULL
   758→   758→    AND trial_end < now();
   759→   759→
   760→   760→  -- Le trigger trg_sync_profile_subscription va automatiquement
   761→   761→  -- synchroniser les profils concernes
   762→   762→
   763→   763→  -- Liberer le verrou
   764→   764→  PERFORM pg_advisory_unlock(54321);
   765→   765→
   766→   766→  RETURN v_count;
   767→   767→END;
   768→   768→$$ LANGUAGE plpgsql SECURITY DEFINER;

   770→   770→-- ---------------------------------------------------------------
   771→   771→-- 9g. get_usage_summary() — Resume de consommation par profil
   772→   772→--      Retourne une ligne par type de limite avec usage courant
   773→   773→-- ---------------------------------------------------------------
   774→   774→CREATE OR REPLACE FUNCTION public.get_usage_summary(
   775→   775→  p_profile_id UUID
   776→   776→) RETURNS TABLE (
   777→   777→  limit_key TEXT,
   778→   778→  limit_value INTEGER,
   779→   779→  limit_label TEXT,
   780→   780→  current_usage BIGINT,
   781→   781→  remaining INTEGER,
   782→   782→  is_unlimited BOOLEAN
   783→   783→) AS $$
   784→   784→BEGIN
   785→   785→  RETURN QUERY
   786→   786→  SELECT
   787→   787→    pl.limit_key,
   788→   788→    pl.limit_value,
   789→   789→    pl.unit_label,
   790→   790→    COALESCE(ur.usage_count, 0) AS current_usage,
   791→   791→    CASE
   792→   792→      WHEN pl.limit_value = -1 THEN -1
   793→   793→      WHEN pl.limit_value IS NULL THEN 0
   794→   794→      ELSE GREATEST((pl.limit_value - COALESCE(ur.usage_count, 0))::INTEGER, 0)
   795→   795→    END AS remaining,
   796→   796→    (pl.limit_value = -1) AS is_unlimited
   797→   797→  FROM public.plan_limits pl
   798→   798→  LEFT JOIN LATERAL (
   799→   799→    SELECT COUNT(*) AS usage_count
   800→   800→    FROM public.usage_records ur2
   801→   801→    WHERE ur2.profile_id = p_profile_id
   802→   802→      AND ur2.usage_type = pl.limit_key
   803→   803→      AND ur2.status IN ('reserved', 'completed')
   804→   804→      AND ur2.created_at >= (
   805→   805→        SELECT COALESCE(
   806→   806→          MIN(s.current_period_start),
   807→   807→          now() - INTERVAL '30 days'
   808→   808→        )
   809→   809→        FROM public.subscriptions s
   810→   810→        WHERE s.profile_id = p_profile_id
   811→   811→          AND s.status IN ('active', 'trialing')
   812→   812→      )
   813→   813→  ) ur ON true
   814→   814→  WHERE pl.plan_id = (
   815→   815→    SELECT COALESCE(
   816→   816→      (SELECT plan_id FROM public.subscriptions
   817→   817→       WHERE profile_id = p_profile_id AND status IN ('active', 'trialing')
   818→   818→       LIMIT 1),
   819→   819→      (SELECT plan FROM public.profiles WHERE id = p_profile_id),
   820→   820→      'starter'
   821→   821→    )
   822→   822→  );
   823→   823→END;
   824→   824→$$ LANGUAGE plpgsql SECURITY DEFINER STABLE;

  1013→  1013→SELECT
  1014→  1014→  p.id AS profile_id,
  1015→  1015→  p.business_name,
  1016→  1016→  p.plan AS current_plan,

  1018→  1018→  p.subscription_start,

  1023→  1023→    WHEN p.subscription_end < now() THEN 0
  1024→  1024→    ELSE GREATEST(EXTRACT(DAY FROM p.subscription_end - now())::INTEGER, 0)

  1042→  1042→  pl.price_monthly,
  1043→  1043→  pl.price_yearly,
