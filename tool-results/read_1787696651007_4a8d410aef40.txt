   311→-- ============================================================
   312→-- 7. AJOUT COLONNES SUBSCRIPTION AU PROFIL
   313→-- ============================================================
   314→-- On ajoute les colonnes au profil existant pour un acces rapide
   315→-- (evite un JOIN avec subscriptions a chaque requete).
   316→-- Ces colonnes sont des COPIES miroirs de l'abonnement actif.
   317→
   318→-- Colonne plan (supprime l'ancienne si elle existe pour la recreer proprement)
   319→DO $$ BEGIN
   320→  IF EXISTS (
   321→    SELECT 1 FROM information_schema.columns
   322→    WHERE table_schema = 'public' AND table_name = 'profiles' AND column_name = 'plan'
   323→  ) THEN
   324→    ALTER TABLE public.profiles DROP COLUMN plan;
   325→  END IF;
   326→END $$;
   327→
   328→ALTER TABLE public.profiles
   329→  ADD COLUMN IF NOT EXISTS plan TEXT DEFAULT 'free'
   330→    CHECK (plan IN ('free', 'starter', 'pro', 'business')),
   331→  ADD COLUMN IF NOT EXISTS subscription_status TEXT DEFAULT 'none'
   332→    CHECK (subscription_status IN ('none', 'active', 'past_due', 'cancelled', 'expired')),
   333→  ADD COLUMN IF NOT EXISTS subscription_start TIMESTAMPTZ,
   334→  ADD COLUMN IF NOT EXISTS subscription_end TIMESTAMPTZ,
   335→  ADD COLUMN IF NOT EXISTS subscription_id UUID REFERENCES public.subscriptions(id) ON DELETE SET NULL,
   336→  ADD COLUMN IF NOT EXISTS chariow_customer_id TEXT;
   337→
   338→-- Renommer stripe_customer_id en chariow_customer_id si l'ancien existe
   339→DO $$ BEGIN
   340→  IF EXISTS (
   341→    SELECT 1 FROM information_schema.columns
   342→    WHERE table_schema = 'public' AND table_name = 'profiles'
   343→      AND column_name = 'stripe_customer_id'
   344→  ) THEN
   345→    ALTER TABLE public.profiles RENAME COLUMN stripe_customer_id TO chariow_customer_id;
   346→  END IF;
   347→END $$;
   348→
   349→COMMENT ON COLUMN public.profiles.plan IS 'Plan actuel (miroir de la subscription active)';
   350→COMMENT ON COLUMN public.profiles.subscription_status IS 'Statut de l\'abonnement actuel';
   351→COMMENT ON COLUMN public.profiles.chariow_customer_id IS 'ID client chez Chariow (si applicable)';

   353→-- ============================================================
   354→-- 8. TRIGGER : Mise a jour miroir profil <-> subscription
   355→-- ============================================================
   356→
   357→-- Quand une subscription est creee/activee, mettre a jour le profil
   358→CREATE OR REPLACE FUNCTION public.sync_profile_subscription()
   359→RETURNS TRIGGER AS $$
   360→DECLARE
   361→  v_plan_name TEXT;
   362→  v_is_active BOOLEAN;
   363→BEGIN
   364→  -- Ne pas propager si on est deja en train de mettre a jour le profil
   365→  -- (evite les boucles infinies avec le trigger ci-dessous)
   366→  IF TG_OP = 'UPDATE' AND OLD.profile_id IS NOT DISTINCT FROM NEW.profile_id
   367→     AND OLD.plan_id IS NOT DISTINCT FROM NEW.plan_id
   368→     AND OLD.status IS NOT DISTINCT FROM NEW.status
   369→     AND OLD.starts_at IS NOT DISTINCT FROM NEW.starts_at
   370→     AND OLD.ends_at IS NOT DISTINCT FROM NEW.ends_at THEN
   371→    RETURN NEW;
   372→  END IF;
   373→
   374→  v_is_active := (NEW.status = 'active' AND NEW.ends_at > now());
   375→
   376→  IF v_is_active THEN
   377→    v_plan_name := COALESCE(
   378→      (SELECT name FROM public.plans WHERE id = NEW.plan_id),
   379→      NEW.plan_name,
   380→      NEW.plan_id
   381→    );
   382→
   383→    UPDATE public.profiles SET
   384→      plan            = NEW.plan_id,
   385→      subscription_status = 'active',
   386→      subscription_start   = NEW.starts_at,
   387→      subscription_end     = NEW.ends_at,
   388→      subscription_id      = NEW.id,
   389→      updated_at           = now()
   390→    WHERE id = NEW.profile_id;
   391→  ELSE
   392→    -- Si la subscription n'est plus active, retrograder le profil
   393→    UPDATE public.profiles SET
   394→      plan            = 'free',
   395→      subscription_status = COALESCE(
   396→        CASE NEW.status
   397→          WHEN 'cancelled' THEN 'cancelled'
   398→          WHEN 'expired'  THEN 'expired'
   399→          WHEN 'upgraded' THEN 'active'  -- Le nouveau plan est deja actif
   400→          WHEN 'downgraded' THEN 'active' -- Le nouveau plan est deja actif
   401→          ELSE 'none'
   402→        END,
   403→        'none'
   404→      ),
   405→      updated_at           = now()
   406→    WHERE id = NEW.profile_id
   407→      AND subscription_id = NEW.id;  -- Ne mettre a jour que si c'est l'abonnement reference
   408→  END IF;
   409→
   410→  RETURN NEW;
   411→END;
   412→$$ LANGUAGE plpgsql SECURITY DEFINER;
   413→
   414→DROP TRIGGER IF EXISTS trg_sync_profile_subscription ON public.subscriptions;
   415→CREATE TRIGGER trg_sync_profile_subscription
   416→  AFTER INSERT OR UPDATE OF status, ends_at, plan_id ON public.subscriptions
   417→  FOR EACH ROW EXECUTE FUNCTION public.sync_profile_subscription();

   419→-- ============================================================
   420→-- 9. TRIGGER : Expiration automatique des abonnements
   421→-- ============================================================
   422→-- Fonction qui expire les abonnements dont la date de fin est passee.
   423→-- A appeler via un cron job (GitHub Actions) ou manuellement.
   424→
   425→CREATE OR REPLACE FUNCTION public.expire_subscriptions()
   426→RETURNS INTEGER AS $$
   427→DECLARE
   428→  v_count INTEGER;
   429→BEGIN
   430→  -- Verrouillage advisory pour eviter les executions concurrentes
   431→  PERFORM pg_advisory_lock(54321);
   432→
   433→  UPDATE public.subscriptions
   434→  SET status = 'expired',
   435→      updated_at = now()
   436→  WHERE status = 'active'
   437→    AND ends_at < now()
   438→  RETURNING COUNT(*) INTO v_count;
   439→
   440→  -- Release le verrou
   441→  PERFORM pg_advisory_unlock(54321);
   442→
   443→  RETURN v_count;
   444→END;
   445→$$ LANGUAGE plpgsql SECURITY DEFINER;
   446→
   447→COMMENT ON FUNCTION public.expire_subscriptions() IS 'Expire les abonnements dont la date de fin est passee. Retourne le nombre d\'abonnements expires.';
   448→
   449→-- Fonction pour verifier et expirer une subscription specifique
   450→CREATE OR REPLACE FUNCTION public.check_subscription_expiry(p_profile_id UUID)
   451→RETURNS void AS $$
   452→DECLARE
   453→  v_sub RECORD;
   454→BEGIN
   455→  SELECT * INTO v_sub
   456→  FROM public.subscriptions
   457→  WHERE profile_id = p_profile_id AND status = 'active'
   458→  ORDER BY ends_at DESC
   459→  LIMIT 1;
   460→
   461→  IF v_sub IS NOT NULL AND v_sub.ends_at < now() THEN
   462→    UPDATE public.subscriptions
   463→    SET status = 'expired', updated_at = now()
   464→    WHERE id = v_sub.id;
   465→
   466→    UPDATE public.profiles
   467→    SET plan = 'free',
   468→        subscription_status = 'expired',
   469→        updated_at = now()
   470→    WHERE id = p_profile_id;
   471→  END IF;
   472→END;
   473→$$ LANGUAGE plpgsql SECURITY DEFINER;

   479→-- Incrementer un compteur d'usage mensuel
   480→CREATE OR REPLACE FUNCTION public.increment_usage(
   481→  p_profile_id UUID,
   482→  p_counter TEXT   -- 'reminders_sent' ou 'appointments_created'
   483→)
   484→RETURNS INTEGER AS $$
   485→DECLARE
   486→  v_month TEXT := to_char(now(), 'YYYY-MM');
   487→  v_new_count INTEGER;
   488→BEGIN
   489→  INSERT INTO public.subscription_usage (profile_id, usage_month, p_counter)
   490→  VALUES (p_profile_id, v_month, 1)
   491→  ON CONFLICT (profile_id, usage_month) DO UPDATE SET
   492→    -- On ne peut pas utiliser un paramètre comme nom de colonne dynamiquement
   493→    -- donc on fait un CASE
   494→    reminders_sent = CASE
   495→      WHEN p_counter = 'reminders_sent' THEN subscription_usage.reminders_sent + 1
   496→      ELSE subscription_usage.reminders_sent
   497→    END,
   498→    appointments_created = CASE
   499→      WHEN p_counter = 'appointments_created' THEN subscription_usage.appointments_created + 1
   500→      ELSE subscription_usage.appointments_created
   501→    END,
   502→    updated_at = now()
   503→  RETURNING
   504→    CASE p_counter
   505→      WHEN 'reminders_sent' THEN subscription_usage.reminders_sent
   506→      ELSE subscription_usage.appointments_created
   507→    END INTO v_new_count;
   508→
   509→  RETURN COALESCE(v_new_count, 0);
   510→EXCEPTION WHEN OTHERS THEN
   511→  RAISE WARNING 'increment_usage failed: %', SQLERRM;
   512→  RETURN -1;
   513→END;
   514→$$ LANGUAGE plpgsql SECURITY DEFINER;

   516→-- Verifier si un utilisateur peut utiliser une fonctionnalite
   517→CREATE OR REPLACE FUNCTION public.can_use_feature(
   518→  p_profile_id UUID,
   519→  p_feature TEXT  -- 'professionals', 'services', 'reminders', 'appointments_per_day', 'clients'
   520→)
   521→RETURNS TABLE(allowed BOOLEAN, current_limit INTEGER, current_count INTEGER, plan_name TEXT) AS $$
   522→DECLARE
   523→  v_plan RECORD;
   524→  v_current_count INTEGER := 0;
   525→  v_limit INTEGER;
   526→  v_month TEXT := to_char(now(), 'YYYY-MM');
   527→BEGIN
   528→  -- Recuperer le plan actuel du profil
   529→  SELECT p.* INTO v_plan
   530→  FROM public.plans p
   531→  JOIN public.profiles pr ON pr.plan = p.id
   532→  WHERE pr.id = p_profile_id;
   533→
   534→  IF NOT FOUND THEN
   535→    -- Profil sans plan = free par defaut
   536→    SELECT * INTO v_plan FROM public.plans WHERE id = 'free';
   537→  END IF;
   538→
   539→  -- Calculer la limite selon la fonctionnalite
   540→  v_limit := CASE p_feature
   541→    WHEN 'professionals'        THEN v_plan.max_professionals
   542→    WHEN 'services'             THEN v_plan.max_services
   543→    WHEN 'reminders'            THEN v_plan.max_reminders_per_month
   544→    WHEN 'appointments_per_day' THEN v_plan.max_appointments_per_day
   545→    WHEN 'clients'              THEN v_plan.max_clients
   546→    ELSE 0
   547→  END;
   548→
   549→  -- Calculer le compteur actuel
   550→  IF p_feature = 'reminders' THEN
   551→    SELECT COALESCE(SUM(reminders_sent), 0) INTO v_current_count
   552→    FROM public.subscription_usage
   553→    WHERE profile_id = p_profile_id AND usage_month = v_month;
   554→  ELSIF p_feature = 'appointments_per_day' THEN
   555→    SELECT COUNT(*) INTO v_current_count
   556→    FROM public.appointments
   557→    WHERE profile_id = p_profile_id
   558→      AND date(starts_at) = CURRENT_DATE
   559→      AND status != 'cancelled';
   560→  ELSIF p_feature = 'services' THEN
   561→    SELECT COUNT(*) INTO v_current_count
   562→    FROM public.services
   563→    WHERE profile_id = p_profile_id AND is_active = true;
   564→  ELSIF p_feature = 'clients' THEN
   565→    SELECT COUNT(*) INTO v_current_count
   566→    FROM public.clients
   567→    WHERE profile_id = p_profile_id;
   568→  ELSIF p_feature = 'professionals' THEN
   569→    -- Pour l'instant, 1 profil = 1 professionnel
   570→    v_current_count := 1;
   571→  END IF;
   572→
   573→  -- -1 = illimite
   574→  allowed := (v_limit = -1) OR (v_current_count < v_limit);
   575→
   576→  RETURN QUERY SELECT allowed, v_limit, v_current_count, v_plan.name;
   577→  RETURN;
   578→END;
   579→$$ LANGUAGE plpgsql SECURITY DEFINER STABLE;
   580→
   581→-- Creer un abonnement (appele par le webhook apres paiement valide)
   582→CREATE OR REPLACE FUNCTION public.activate_subscription(
   583→  p_profile_id UUID,
   584→  p_plan_id    TEXT,
   585→  p_payment_id UUID DEFAULT NULL,
   586→  p_duration_days INTEGER DEFAULT 30
   587→)
   588→RETURNS UUID AS $$
   589→DECLARE
   590→  v_existing_active UUID;
   591→  v_new_sub_id UUID;
   592→  v_old_status TEXT;
   593→BEGIN
   594→  -- Verrouillage advisory par profil (evite les doubles activations)
   595→  PERFORM pg_advisory_lock(hashtext(p_profile_id::text));
   596→
   597→  -- Trouver l'abonnement actuel s'il existe
   598→  SELECT id INTO v_existing_active
   599→  FROM public.subscriptions
   600→  WHERE profile_id = p_profile_id AND status = 'active';
   601→
   602→  IF v_existing_active IS NOT NULL THEN
   603→    -- Determiner si c'est un upgrade ou downgrade
   604→    SELECT
   605→      CASE
   606→        WHEN (SELECT tier_priority FROM plans WHERE id = p_plan_id) >
   607→             (SELECT tier_priority FROM plans WHERE id = s.plan_id)
   608→        THEN 'upgraded'
   609→        ELSE 'downgraded'
   610→      END INTO v_old_status
   611→    FROM public.subscriptions s WHERE id = v_existing_active;
   612→
   613→    -- Desactiver l'ancien abonnement
   614→    UPDATE public.subscriptions
   615→    SET status = v_old_status, updated_at = now()
   616→    WHERE id = v_existing_active;
   617→  END IF;
   618→
   619→  -- Creer le nouvel abonnement
   620→  INSERT INTO public.subscriptions (
   621→    profile_id, plan_id, plan_name,
   622→    starts_at, ends_at,
   623→    status, payment_id
   624→  ) VALUES (
   625→    p_profile_id, p_plan_id,
   626→    COALESCE((SELECT name FROM plans WHERE id = p_plan_id), p_plan_id),
   627→    now(),
   628→    now() + (p_duration_days || ' days')::INTERVAL,
   629→    'active',
   630→    p_payment_id
   631→  )
   632→  RETURNING id INTO v_new_sub_id;
   633→
   634→  -- Le trigger trg_sync_profile_subscription va automatiquement
   635→  -- mettre a jour les colonnes du profil
   636→
   637→  -- Release le verrou
   638→  PERFORM pg_advisory_unlock(hashtext(p_profile_id::text));
   639→
   640→  RETURN v_new_sub_id;
   641→END;
   642→$$ LANGUAGE plpgsql SECURITY DEFINER;
   643→
   644→COMMENT ON FUNCTION public.activate_subscription() IS 'Active un abonnement apres paiement. Desactive l\'ancien si existant. Verrouille par profil pour eviter les doubles activations.';

   889→-- Vue : resume abonnement d'un profil
   890→CREATE OR REPLACE VIEW public.v_subscription_summary AS
   891→SELECT
   892→  p.id AS profile_id,
   893→  p.plan,
   894→  p.subscription_status,
   895→  p.subscription_start,
   896→  p.subscription_end,
   897→  pl.name        AS plan_name,
   898→  pl.price_monthly,
   899→  pl.features,
   900→  CASE
   901→    WHEN p.subscription_status = 'active' AND p.subscription_end > now()
   902→    THEN GREATEST(0, CEIL(EXTRACT(EPOCH FROM (p.subscription_end - now())) / 86400))
   903→    ELSE 0
   904→  END AS days_remaining,
   905→  CASE
   906→    WHEN p.subscription_status = 'active' AND p.subscription_end > now() THEN true
   907→    ELSE false
   908→  END AS is_active,
   909→  -- Dernier paiement
   910→  (SELECT status FROM public.payments
   911→   WHERE profile_id = p.id AND status = 'completed'
   912→   ORDER BY paid_at DESC LIMIT 1) AS last_payment_status,
   913→  (SELECT paid_at FROM public.payments
   914→   WHERE profile_id = p.id AND status = 'completed'
   915→   ORDER BY paid_at DESC LIMIT 1) AS last_payment_date,
   916→  -- Usage ce mois
   917→  COALESCE(u.reminders_sent, 0) AS reminders_used_this_month,
   918→  COALESCE(u.appointments_created, 0) AS appointments_this_month
   919→FROM public.profiles p
   920→LEFT JOIN public.plans pl ON pl.id = p.plan
   921→LEFT JOIN public.subscription_usage u
   922→  ON u.profile_id = p.id AND u.usage_month = to_char(now(), 'YYYY-MM');
   923→
   924→-- Vue : revenus mensuels (pour le dashboard admin)
   925→CREATE OR REPLACE VIEW public.v_monthly_revenue AS
   926→SELECT
   927→  to_char(paid_at, 'YYYY-MM') AS month,
   928→  plan_id,
   929→  plan_name,
   930→  COUNT(*) AS payment_count,
   931→  SUM(amount) AS total_amount,
   932→  currency
   933→FROM public.payments
   934→WHERE status = 'completed' AND paid_at IS NOT NULL
   935→GROUP BY to_char(paid_at, 'YYYY-MM'), plan_id, plan_name, currency
   936→ORDER BY month DESC, total_amount DESC;
   937→
   938→-- Vue : abonnements expirant bientot (dans les 7 prochains jours)
   939→CREATE OR REPLACE VIEW public.v_expiring_soon AS
   940→SELECT
   941→  s.id AS subscription_id,
   942→  s.profile_id,
   943→  s.plan_id,
   944→  s.plan_name,
   945→  s.ends_at,
   946→  p.business_name,
   947→  p.email,
   948→  GREATEST(0, CEIL(EXTRACT(EPOCH FROM (s.ends_at - now())) / 86400)) AS days_left
   949→FROM public.subscriptions s
   950→JOIN public.profiles p ON p.id = s.profile_id
   951→WHERE s.status = 'active'
   952→  AND s.ends_at > now()
   953→  AND s.ends_at < now() + INTERVAL '7 days'
