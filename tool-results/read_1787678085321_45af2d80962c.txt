    66→function formatCurrency(amount: number): string {
    67→  return new Intl.NumberFormat('fr-FR', { style: 'currency', currency: 'XAF', minimumFractionDigits: 0 }).format(amount);
    68→}
    69→
    70→function formatDate(iso: string): string {
    71→  return new Intl.DateTimeFormat('fr-FR', { day: '2-digit', month: 'short', hour: '2-digit', minute: '2-digit' }).format(new Date(iso));
    72→}
    73→
    74→function statusBadge(status: string) {
    75→  const map: Record<string, { label: string; cls: string }> = {
    76→    pending:    { label: 'En attente',  cls: 'bg-amber-100 text-amber-700 dark:bg-amber-900/40 dark:text-amber-400' },
    77→    confirmed:  { label: 'Confirmé',   cls: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/40 dark:text-emerald-400' },
    78→    completed:  { label: 'Terminé',    cls: 'bg-blue-100 text-blue-700 dark:bg-blue-900/40 dark:text-blue-400' },
    79→    cancelled:  { label: 'Annulé',     cls: 'bg-red-100 text-red-700 dark:bg-red-900/40 dark:text-red-400' },
    80→    no_show:    { label: 'Absent',     cls: 'bg-orange-100 text-orange-700 dark:bg-orange-900/40 dark:text-orange-400' },
    81→  };
    82→  const cfg = map[status] || { label: status, cls: 'bg-gray-100 text-gray-700' };
    83→  return <Badge variant="secondary" className={cfg.cls}>{cfg.label}</Badge>;
    84→}
    85→
    86→function GrowthIndicator({ value, label }: { value: number; label: string }) {
    87→  if (value === 0) return <span className="text-xs text-muted-foreground">{label}</span>;
    88→  const isUp = value > 0;
    89→  return (
    90→    <span className={`inline-flex items-center gap-0.5 text-xs font-medium ${isUp ? 'text-emerald-600 dark:text-emerald-400' : 'text-red-600 dark:text-red-400'}`}>
    91→      {isUp ? <ArrowUpRight size={14} /> : <ArrowDownRight size={14} />}
    92→      {Math.abs(value)}% {label}
    93→    </span>
    94→  );
    95→}
